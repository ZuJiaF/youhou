    async function waitForSelector(selector, maxWait = 30000) {
        let wait = 200;
        let total = 0;
        while (total < maxWait) {
            const el = document.querySelector(selector);
            if (el) return el;
            await new Promise(r => setTimeout(r, wait));
            total += wait;
            wait = Math.min(wait * 2, 5000);
        }
        return null;
    }

    // 等待指定容器的DOM停止变化（连续0.5秒无变化视为稳定）
    function waitForDomStable(selector, quietMs = 500, maxWait = 15000) {
        return new Promise(resolve => {
            const target = document.querySelector(selector);
            if (!target) { resolve(); return; }

            let timer = null;
            let totalTimer = null;

            const obs = new MutationObserver(() => {
                clearTimeout(timer);
                timer = setTimeout(() => {
                    obs.disconnect();
                    clearTimeout(totalTimer);
                    resolve();
                }, quietMs);
            });

            obs.observe(target, { childList: true, subtree: true });

            // 初始启动计时
            timer = setTimeout(() => {
                obs.disconnect();
                clearTimeout(totalTimer);
                resolve();
            }, quietMs);

            // 最大等待时间兜底
            totalTimer = setTimeout(() => {
                obs.disconnect();
                clearTimeout(timer);
                resolve();
            }, maxWait);
        });
    }

    function queryFirst(root, selectors) {
        for (const selector of selectors) {
            const element = root.querySelector(selector);
            if (element) return element;
        }
        return null;
    }

    function findRowSkuElement(row) {
        return queryFirst(row, RECEIVE_SKU_SELECTORS);
    }

    function findRowReceiveInput(row) {
        const preferredInput = queryFirst(
            row, RECEIVE_INPUT_SELECTORS
        );
        if (preferredInput) return preferredInput;

        return Array.from(row.querySelectorAll('input'))
            .find(input => ![
                'button', 'checkbox', 'hidden',
                'radio', 'reset', 'submit'
            ].includes(input.type) && !input.disabled) || null;
    }

    function normalizeSku(text) {
        return String(text || '')
            .trim()
            .replace(/^SKU:\s*/i, '');
    }

    function rowTextContainsSku(rowText, sku) {
        let startIndex = 0;

        while (startIndex < rowText.length) {
            const index = rowText.indexOf(sku, startIndex);
            if (index === -1) return false;

            const before = rowText[index - 1] || '';
            const after = rowText[index + sku.length] || '';
            const isSkuChar = char =>
                /[a-zA-Z0-9._-]/.test(char);

            if (!isSkuChar(before) && !isSkuChar(after)) {
                return true;
            }

            startIndex = index + sku.length;
        }

        return false;
    }

    function getRowSku(row, skuMap) {
        const skuElement = findRowSkuElement(row);
        const explicitSku = normalizeSku(
            skuElement && skuElement.innerText
        );

        if (explicitSku && explicitSku in skuMap) {
            return explicitSku;
        }

        const rowText = row.innerText || row.textContent || '';
        return Object.keys(skuMap)
            .sort((a, b) => b.length - a.length)
            .find(sku => rowTextContainsSku(rowText, sku)) || '';
    }

    function inspectReceiveTable(table) {
        const rows = Array.from(
            table.querySelectorAll('tbody > tr')
        );
        let skuRows = 0;
        let inputRows = 0;
        let pairedRows = 0;

        rows.forEach(row => {
            const hasSku = Boolean(findRowSkuElement(row));
            const hasInput = Boolean(findRowReceiveInput(row));
            if (hasSku) skuRows++;
            if (hasInput) inputRows++;
            if (hasSku && hasInput) pairedRows++;
        });

        const tableText =
            table.innerText || table.textContent || '';
        const hasShelfMarker = tableText.includes('添加货架');
        const score =
            (hasShelfMarker ? 100000 : 0) +
            pairedRows * 10000 +
            skuRows * 1000 +
            inputRows * 100 +
            rows.length;

        return {
            table,
            rows,
            skuRows,
            inputRows,
            pairedRows,
            hasShelfMarker,
            score
        };
    }

    function findReceiveTable() {
        const candidates = [];
        const seen = new Set();
        const addCandidate = table => {
            if (!seen.has(table)) {
                seen.add(table);
                candidates.push(table);
            }
        };

        RECEIVE_TABLE_SELECTORS.forEach(selector => {
            document.querySelectorAll(selector)
                .forEach(addCandidate);
        });

        // BigSeller 新版采购收货页的明细 table 不再带 class。
        document.querySelectorAll('table').forEach(addCandidate);

        const ranked = candidates
            .map(inspectReceiveTable)
            .filter(info =>
                info.rows.length > 0 &&
                info.inputRows > 0 &&
                (
                    info.hasShelfMarker ||
                    info.pairedRows > 0 ||
                    info.skuRows > 0
                )
            )
            .sort((a, b) => b.score - a.score);

        // [debug-done][2026-08-11] 排查：新版收货明细表无 class，旧选择器未命中。
        // console.log('[debug][2026-08-11][findReceiveTable] 候选表格:', ranked);
        return ranked[0] || null;
    }

    document.getElementById('fillBtn').onclick = () => {

            .getElementById('skuList')
            .value
            .split('\n')

        const qtyArr = document
            .getElementById('qtyList')
            .value
            .split('\n')
            .map(v => v.trim())
            .filter(Boolean);

            alert(
                `SKU数量(${skuArr.length}) 与 数量行数(${qtyArr.length}) 不一致`
            );

            return;
        }

        const skuMap = {};

        skuArr.forEach((sku, index) => {
            skuMap[sku] = qtyArr[index];
        });

        const tableInfo = findReceiveTable();

        if (!tableInfo) {

            alert("未找到表格");

            return;
        }

        let success = 0;

        const rows = tableInfo.rows;

        rows.forEach(row => {

            const sku = getRowSku(row, skuMap);

            if (!(sku in skuMap))
                return;

            const input = findRowReceiveInput(row);

            if (!input)
                return;

            const value =
                skuMap[sku];

            input.focus();

            // 兼容 React 受控组件的值设置
            const nativeInputValueSetter =
                Object.getOwnPropertyDescriptor(
                    window.HTMLInputElement.prototype,
                    'value'
                ).set;

            nativeInputValueSetter.call(
                input, value
            );

            input.dispatchEvent(
                new Event('input', {
                    bubbles: true
                })
            );

            input.dispatchEvent(
                new Event('change', {
                    bubbles: true
                })
            );

            input.dispatchEvent(
                new KeyboardEvent(
                    'keyup',
                    {
                        bubbles: true
                    }
                )
            );

            input.blur();

            row.style.background =
                'rgba(0,255,0,.15)';

            success++;
        });

            .innerText =
            `成功填写 ${success} 个SKU`;

                const json = await resp.json();
                const responseRows = json.data?.page?.rows || [];

                console.log(
                    `[debug][2026-07-24][queryInventory] 第 ${i + 1} 批响应`,
                    JSON.stringify({
                        httpStatus: resp.status,
                        code: json.code,
                        message: json.message || json.msg || '',
                        warehouseId: group.warehouseId,
                        requestSkus: batch,
                        rowCount: responseRows.length,
                        firstRowFields: responseRows[0] ? Object.keys(responseRows[0]) : [],
                        responseRows: responseRows.map(item => ({
                            sku: item.sku,
                            skuName: item.skuName,
                            warehouseId: item.warehouseId,
                            warehouseName: item.warehouseName
                        }))
                    })
                );

                if (json.code === 0 && json.data && json.data.page) {
                    json.data.page.rows.forEach(item => {
                        const responseSku = item.sku || item.skuName;
                        if (!responseSku) return;

                        skuDataMap[inventoryKey(group.warehouseId, responseSku)] = {
                            shelfName: item.shelfName || '--',
                            onhand: item.onhand ?? '--',
                            available: item.available ?? '--'
                        };
                    });
                }
            }
        } catch (e) {
            msgEl.innerText = `查询出错: ${e.message}`;
            return;
        }

        console.log(
            '[debug][2026-07-24][queryInventory] 匹配汇总',
            JSON.stringify({
                requestedCount: skuRecords.filter(record => record.warehouse).length,
                mappedCount: Object.keys(skuDataMap).length,
                mappedSkus: Object.keys(skuDataMap),
                unmatchedSkus: skuRecords
                    .filter(record => record.warehouse)
                    .filter(record => !skuDataMap[inventoryKey(record.warehouse.id, record.sku)])
                    .map(record => ({
                        packageNo: record.packageNo,
                        warehouseId: record.warehouse.id,
                        sku: record.sku
                    }))
            })
        );

        // 注入标签到页面
        let injected = 0;
        skuRecords.forEach(record => {
            const { span, sku, warehouse, isStockout } = record;
            if (!warehouse) return;

            const info = skuDataMap[inventoryKey(warehouse.id, sku)];
            if (!info) return;

            const onhand = Number(info.onhand);
            const needsPhysicalCheck =
                isStockout && Number.isFinite(onhand) && onhand > STOCK_WARNING_THRESHOLD;

            // 插到span.ml_10的直接父div中
            const parent = span.parentElement;
            if (parent.querySelector('.c_inv_tag')) return;

            const tag = document.createElement('div');
            tag.className = 'c_inv_tag';
            tag.style.cssText = needsPhysicalCheck
                ? 'display:inline-block;max-width:190px;margin-left:10px;margin-top:2px;padding:4px 7px;border:1px solid #ff4d4f;border-radius:3px;font-size:12px;background:#fff1f0;color:#cf1322;line-height:1.5;white-space:normal;'
                : 'display:inline-block;margin-left:10px;margin-top:2px;padding:1px 6px;border-radius:3px;font-size:12px;background:#e6f7ff;color:#1890ff;line-height:1.5;';
            tag.innerHTML =
                `${info.shelfName}<br>现有库存:${info.onhand}<br>整仓可用:${info.available}` +
                (needsPhysicalCheck
                    ? '<br><strong>库存较多，疑似误报<br>请主管实地检查</strong>'
                    : '');
            parent.appendChild(tag);
            injected++;
        });

        const notFoundKeys = new Set(
            skuRecords
                .filter(record => record.warehouse)
                .filter(record => !skuDataMap[inventoryKey(record.warehouse.id, record.sku)])
                .map(record => inventoryKey(record.warehouse.id, record.sku))
        );
        const missingWarehouseCount = missingWarehousePackages.size;

        msgEl.innerText = `完成！已标注 ${injected} 处` +
            (notFoundKeys.size > 0 ? `，${notFoundKeys.size} 个仓库SKU未查到` : '') +
            (missingWarehouseCount > 0 ? `，${missingWarehouseCount} 个包裹未识别仓库` : '');

        // 兜底：查询完成后注入换货按钮（防重复逻辑在函数内部）
        injectReplaceGoodsBtn();
    };

    autoQueryWhenReady();

    async function injectReplaceGoodsBtn() {
        if (!isInventoryPage()) return;

        // 等待包裹标题栏加载
        await waitForSelector('#pageList .items_tit');

        // 在每个包裹标题栏左侧区域注入按钮
        const titles = document.querySelectorAll('#pageList .items_tit');
        titles.forEach(tit => {
            if (tit.querySelector('.c_replace_btn')) return;

            const lt = tit.querySelector('.lt') || tit;

            const btn = document.createElement('button');
            btn.className = 'c_replace_btn';
            btn.innerText = '查询是否换货';
            btn.style.cssText =
                'margin-left:10px;padding:2px 10px;border:1px solid #faad14;background:#fffbe6;color:#d48806;border-radius:4px;cursor:pointer;font-size:12px;white-space:nowrap;';
            btn.onclick = () => queryReplaceForPackage(btn, tit);
            lt.appendChild(btn);
        });

        injectWarehouseTags();
    }

    function getPackageNoFromTitle(titleEl) {
        const lt = titleEl.querySelector('.lt') || titleEl;
        const match = lt.innerText.match(/\bBS[A-Z0-9]+\b/i);
        return match ? match[0].toUpperCase() : '';
    }

    function injectWarehouseTags() {
        const titles = document.querySelectorAll('#pageList .items_tit');
        titles.forEach(tit => {
            const packageNo = getPackageNoFromTitle(tit);
            const warehouse = packageWarehouseCache[packageNo];
            if (!warehouse) return;

            const lt = tit.querySelector('.lt') || tit;
            let tag = lt.querySelector('.c_warehouse_tag');
            if (!tag) {
                tag = document.createElement('span');
                tag.className = 'c_warehouse_tag';
                tag.style.cssText =
                    'margin-left:8px;padding:2px 8px;border:1px solid #91caff;background:#e6f4ff;color:#0958d9;border-radius:3px;font-size:12px;white-space:nowrap;vertical-align:middle;';

                const replaceBtn = lt.querySelector('.c_replace_btn');
                if (replaceBtn) replaceBtn.after(tag);
                else lt.appendChild(tag);
            }

            tag.innerText = `仓库：${warehouse.name}`;
            tag.title = `仓库ID：${warehouse.id}`;
        });
    }

    // 自动收集订单号并批量获取ID（与SKU查询同时进行）
    async function autoFetchOrderIds() {
        if (!isInventoryPage()) return;

        const tdSel = '#pageList .list_items tbody tr td.order_No, #pageList .list_items tbody tr td:nth-child(5)';
        await waitForSelector(tdSel);

        const orderTds = document.querySelectorAll(tdSel);
        const orderNos = [];
        orderTds.forEach(td => {
            const text = td.innerText.replace(/\s+/g, '').trim();
            if (text) {
                orderNos.push(text);
            }
        });

        if (!orderNos.length) return;

        try {
            orderIdCache = {};
            packageWarehouseCache = {};

            const uniqueOrderNos = [...new Set(orderNos)];
            const orderBatches = [];
            for (let i = 0; i < uniqueOrderNos.length; i += BATCH_SIZE) {
                orderBatches.push(uniqueOrderNos.slice(i, i + BATCH_SIZE));
            }

            for (const batch of orderBatches) {
                const resp = await fetch('/api/v1/order/new/pageList.json', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                        'clienttype': '1'
                    },
                    body: JSON.stringify({
                        searchType: 'orderNo',
                        searchContent: batch.join(','),
                        inquireType: 2,
                        allOrder: true,
                        historyOrder: 0,
                        pageNo: 1,
                        pageSize: 300,
                        desc: 1,
                        orderBy: 'printTime'
                    })
                });

                const json = await resp.json();
                if (json.code !== 0 || !json.data?.page?.rows) continue;

                json.data.page.rows.forEach(order => {
                    if (order.platformOrderId) {
                        orderIdCache[order.platformOrderId] = order.id;
                    }
                    if (order.packageNo && order.warehouseId) {
                        packageWarehouseCache[String(order.packageNo).toUpperCase()] = {
                            id: String(order.warehouseId),
                            name: order.shipmentWarehouse || `仓库 ${order.warehouseId}`
                        };
                    }
                });
            }

            console.log(
                `[c仓助手] 已缓存 ${Object.keys(orderIdCache).length} 个订单ID，` +
                `${Object.keys(packageWarehouseCache).length} 个包裹仓库`
            );
            injectWarehouseTags();
        } catch (e) {
            console.log('[c仓助手] 获取订单ID失败:', e.message);
        }
    }

    // 点击单个包裹的按钮：用缓存的ID查该订单详情
    async function queryReplaceForPackage(btn, titleEl) {
        const origText = btn.innerText;
        btn.innerText = '查询中...';
        btn.disabled = true;

        try {
            // 找到该包裹对应的表格
            const listItemsOut = titleEl.closest('.list_items_out');
            if (!listItemsOut) return;

            // 获取订单号
            const orderTd = listItemsOut.querySelector(
                'table.list_items tbody tr td.order_No, table.list_items tbody tr td:nth-child(5)'
            );
            if (!orderTd) return;

            const orderNo = orderTd.innerText.trim();
            if (!orderNo || !/^\d+$/.test(orderNo)) {
                btn.innerText = '无订单号';
                setTimeout(() => { btn.innerText = origText; }, 2000);
                return;
            }

            // 从缓存获取订单ID，若无则单独查询
            let orderId = orderIdCache[orderNo];
            if (!orderId) {
                const resp1 = await fetch('/api/v1/order/new/pageList.json', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                        'clienttype': '1'
                    },
                    body: JSON.stringify({
                        searchType: 'orderNo',
                        searchContent: orderNo,
                        inquireType: 2,
                        allOrder: true,
                        historyOrder: 0,
                        pageNo: 1,
                        pageSize: 10,
                        desc: 1,
                        orderBy: 'printTime'
                    })
                });
                const json1 = await resp1.json();
                if (json1.code === 0 && json1.data && json1.data.page && json1.data.page.rows.length) {
                    orderId = json1.data.page.rows[0].id;
                    orderIdCache[orderNo] = orderId;
                }
            }

            if (!orderId) {
                btn.innerText = '未找到订单';
                setTimeout(() => { btn.innerText = origText; }, 2000);
                return;
            }

            // 查详情
            const resp2 = await fetch(
                `/api/v1/order/detail.json?id=${orderId}&historyOrder=0&viewBuyerMessage=false`,
                { headers: { 'clienttype': '1' } }
            );
            const json2 = await resp2.json();

            if (json2.code !== 0 || !json2.data) {
                btn.innerText = '查询失败';
                setTimeout(() => { btn.innerText = origText; }, 2000);
                return;
            }

            const detail = json2.data.orderDetail;
            const replaces = [];

            if (detail.orderItemVoList) {
                detail.orderItemVoList.forEach(item => {
                    if (item.inventorySku && item.inventorySku !== item.varSku) {
                        replaces.push({
                            from: item.varSku,
                            to: item.inventorySku
                        });
                    }
                });
            }

            if (replaces.length > 0) {
                btn.innerText = '有换货';
                btn.style.background = '#fff1f0';
                btn.style.borderColor = '#cf1322';
                btn.style.color = '#cf1322';

                // 在标题栏显示换货详情
                let tag = titleEl.querySelector('.c_replace_tag');
                if (!tag) {
                    tag = document.createElement('span');
                    tag.className = 'c_replace_tag';
                    tag.style.cssText =
                        'margin-left:8px;padding:2px 6px;border-radius:3px;font-size:11px;background:#fff1f0;color:#cf1322;';
                    titleEl.appendChild(tag);
                }
                tag.innerHTML = replaces.map(
                    r => `${r.from} → ${r.to}`
                ).join(', ');
            } else {
                btn.innerText = '无换货';
                btn.style.background = '#f6ffed';
                btn.style.borderColor = '#52c41a';
                btn.style.color = '#52c41a';
            }

        } catch (e) {
            btn.innerText = '出错';
            setTimeout(() => { btn.innerText = origText; }, 2000);
        } finally {
            btn.disabled = false;
        }
    }

    injectReplaceGoodsBtn();
    autoFetchOrderIds();

})();
