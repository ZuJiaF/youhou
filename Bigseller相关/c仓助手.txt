    async function waitForSelector(selector, maxWait = 30000) {
        let wait = 200;
        let total = 0;
        while (total < maxWait) {
            const el = document.querySelector(selector);
            if (el) return el;
            await new Promise(r => setTimeout(r, wait));
            total += wait;
            wait = Math.min(wait * 2, 5000);
        }
        return null;
    }

    // 等待指定容器的DOM停止变化（连续0.5秒无变化视为稳定）
    function waitForDomStable(selector, quietMs = 500, maxWait = 15000) {
        return new Promise(resolve => {
            const target = document.querySelector(selector);
            if (!target) { resolve(); return; }

            let timer = null;
            let totalTimer = null;

            const obs = new MutationObserver(() => {
                clearTimeout(timer);
                timer = setTimeout(() => {
                    obs.disconnect();
                    clearTimeout(totalTimer);
                    resolve();
                }, quietMs);
            });

            obs.observe(target, { childList: true, subtree: true });

            // 初始启动计时
            timer = setTimeout(() => {
                obs.disconnect();
                clearTimeout(totalTimer);
                resolve();
            }, quietMs);

            // 最大等待时间兜底
            totalTimer = setTimeout(() => {
                obs.disconnect();
                clearTimeout(timer);
                resolve();
            }, maxWait);
        });
    }

    function getReceiveRectInfo(element) {
        if (!element) return null;

        const rect = element.getBoundingClientRect();
        return {
            top: Math.round(rect.top),
            bottom: Math.round(rect.bottom),
            left: Math.round(rect.left),
            right: Math.round(rect.right),
            width: Math.round(rect.width),
            height: Math.round(rect.height),
            visibleInViewport:
                rect.width > 0 &&
                rect.height > 0 &&
                rect.bottom > 0 &&
                rect.top < window.innerHeight
        };
    }

    function getReceiveVirtualInfo(table) {
        const containers = [];
        let current = table && table.parentElement;

        while (current && current !== document.body) {
            const style = window.getComputedStyle(current);
            const hasScrollableOverflow =
                current.scrollHeight > current.clientHeight + 2 ||
                current.scrollWidth > current.clientWidth + 2 ||
                /(auto|scroll|overlay)/.test(
                    `${style.overflow} ${style.overflowY} ${style.overflowX}`
                );

            if (hasScrollableOverflow) {
                containers.push({
                    tagName: current.tagName,
                    id: current.id || '',
                    className: String(current.className || '')
                        .slice(0, 180),
                    overflow: style.overflow,
                    overflowY: style.overflowY,
                    scrollTop: current.scrollTop,
                    scrollLeft: current.scrollLeft,
                    clientHeight: current.clientHeight,
                    scrollHeight: current.scrollHeight,
                    clientWidth: current.clientWidth,
                    scrollWidth: current.scrollWidth,
                    childCount: current.children.length,
                    rect: getReceiveRectInfo(current)
                });
            }

            current = current.parentElement;
        }

        let virtualRoot = table && table.parentElement;
        let virtualMarkers = [];

        while (virtualRoot && virtualRoot !== document.body) {
            const found = Array.from(
                virtualRoot.querySelectorAll(
                    '.bs_newvirtual-blank, ' +
                    '[class*="virtual-list"], ' +
                    '[class*="virtual-scroll"]'
                )
            );

            if (found.length > 0) {
                virtualMarkers = found.slice(0, 20).map(marker => ({
                    tagName: marker.tagName,
                    className: String(marker.className || '')
                        .slice(0, 180),
                    style: marker.getAttribute('style') || '',
                    clientHeight: marker.clientHeight,
                    scrollHeight: marker.scrollHeight,
                    rect: getReceiveRectInfo(marker)
                }));
                break;
            }

            virtualRoot = virtualRoot.parentElement;
        }

        return {
            tableRect: getReceiveRectInfo(table),
            containers,
            virtualRoot: virtualRoot ? {
                tagName: virtualRoot.tagName,
                id: virtualRoot.id || '',
                className: String(virtualRoot.className || '')
                    .slice(0, 180)
            } : null,
            virtualMarkers
        };
    }

    function queryFirst(root, selectors) {
        for (const selector of selectors) {
            const element = root.querySelector(selector);
            if (element) return element;
        }
        return null;
    }

    function findRowSkuElement(row) {
        return queryFirst(row, RECEIVE_SKU_SELECTORS);

    function findRowReceiveInput(row) {
        const preferredInput = queryFirst(
            row, RECEIVE_INPUT_SELECTORS
        );
        if (preferredInput) return preferredInput;

        return Array.from(row.querySelectorAll('input'))
            .find(input => ![
                'button', 'checkbox', 'hidden',
                'radio', 'reset', 'submit'
            ].includes(input.type) && !input.disabled) || null;
    }

    function normalizeSku(text) {
        return String(text || '')
            .trim()
            .replace(/^SKU:\s*/i, '');
    }

    function rowTextContainsSku(rowText, sku) {
        let startIndex = 0;

        while (startIndex < rowText.length) {
            const index = rowText.indexOf(sku, startIndex);
            if (index === -1) return false;

            const before = rowText[index - 1] || '';
            const after = rowText[index + sku.length] || '';
            const isSkuChar = char =>
                /[a-zA-Z0-9._-]/.test(char);

            if (!isSkuChar(before) && !isSkuChar(after)) {
                return true;
            }

            startIndex = index + sku.length;
        }

        return false;
    }

    function getRowSku(row, skuMap) {
        const skuElement = findRowSkuElement(row);
        const explicitSku = normalizeSku(
            skuElement && skuElement.innerText
        );

        if (explicitSku && explicitSku in skuMap) {
            return explicitSku;
        }

        const rowText = row.innerText || row.textContent || '';
        return Object.keys(skuMap)
            .sort((a, b) => b.length - a.length)
            .find(sku => rowTextContainsSku(rowText, sku)) || '';
    }

    function inspectReceiveTable(table) {
        const rows = Array.from(
            table.querySelectorAll('tbody > tr')
        );
        let skuRows = 0;
        let inputRows = 0;
        let pairedRows = 0;

        rows.forEach(row => {
            const hasSku = Boolean(findRowSkuElement(row));
            const hasInput = Boolean(findRowReceiveInput(row));
            if (hasSku) skuRows++;
            if (hasInput) inputRows++;
            if (hasSku && hasInput) pairedRows++;
        });

        const tableText =
            table.innerText || table.textContent || '';
        const hasShelfMarker = tableText.includes('添加货架');
        const score =
            (hasShelfMarker ? 100000 : 0) +
            pairedRows * 10000 +
            skuRows * 1000 +
            inputRows * 100 +
            rows.length;

        return {
            table,
            rows,
            skuRows,
            inputRows,
            pairedRows,
            hasShelfMarker,
            score
        };
    }

    function findReceiveTable() {
        const candidates = [];
        const seen = new Set();
        const addCandidate = table => {
            if (!seen.has(table)) {
                seen.add(table);
                candidates.push(table);
            }
        };

        RECEIVE_TABLE_SELECTORS.forEach(selector => {
            document.querySelectorAll(selector)
                .forEach(addCandidate);
        });

        // BigSeller 新版采购收货页的明细 table 不再带 class。
        document.querySelectorAll('table').forEach(addCandidate);

        const ranked = candidates
            .map(inspectReceiveTable)
            .filter(info =>
                info.rows.length > 0 &&
                info.inputRows > 0 &&
                (
                    info.hasShelfMarker ||
                    info.pairedRows > 0 ||
                    info.skuRows > 0
                )
            )
            .sort((a, b) => b.score - a.score);

        // [debug-done][2026-08-11] 排查：新版收货明细表无 class，旧选择器未命中。
        // console.log('[debug][2026-08-11][findReceiveTable] 候选表格:', ranked);
        receiveDebug('findReceiveTable', {
            candidateCount: candidates.length,
            ranked: ranked.map(info => ({
                tableIndex: Array.from(
                    document.querySelectorAll('table')
                ).indexOf(info.table),
                rows: info.rows.length,
                skuRows: info.skuRows,
                inputRows: info.inputRows,
                pairedRows: info.pairedRows,
                hasShelfMarker: info.hasShelfMarker,
                score: info.score,
                className: String(info.table.className || ''),
                virtual: getReceiveVirtualInfo(info.table)
            }))
        });
        return ranked[0] || null;
    }

    function getReceiveRowDebug(row, rowIndex, skuArr, input) {
        const rowText = (
            row.innerText || row.textContent || ''
        ).replace(/\s+/g, ' ').trim();
        const skuElement = findRowSkuElement(row);
        const explicitSku = normalizeSku(
            skuElement && skuElement.innerText
        );
        const inputSelector = input
            ? RECEIVE_INPUT_SELECTORS.find(
                selector => row.querySelector(selector) === input
            ) || 'fallback-input'
            : '';

        return {
            rowIndex,
            dataRowIndex: row.getAttribute('data-rowindex') || '',
            className: String(row.className || ''),
            connected: row.isConnected,
            rect: getReceiveRectInfo(row),
            explicitSku,
            textHitTargets: skuArr.filter(
                sku => rowTextContainsSku(rowText, sku)
            ),
            selectedInput: input ? {
                selector: inputSelector,
                type: input.type,
                value: input.value,
                disabled: input.disabled,
                connected: input.isConnected,
                className: String(input.className || ''),
                rect: getReceiveRectInfo(input)
            } : null,
            allInputs: Array.from(
                row.querySelectorAll('input')
            ).map(item => ({
                type: item.type,
                value: item.value,
                disabled: item.disabled,
                connected: item.isConnected,
                className: String(item.className || ''),
                rect: getReceiveRectInfo(item)
            })),
            rowText: rowText.slice(0, 320)
        };
    }

                `SKU数量(${skuArr.length}) 与 数量行数(${qtyArr.length}) 不一致`
            );

            return;
        }

        // [debug][2026-08-12] 检测两个输入框内容是否完全一样（用户误填）
        const skuArrStr = skuArr.join('|');
        const qtyArrStr = qtyArr.join('|');
        if (skuArrStr === qtyArrStr) {
            receiveDebug('fill-input-duplicate', {
                message: 'SKU列表和数量列表内容完全相同',
                sample: skuArr.slice(0, 3)
            });

            alert('两个输入框内容完全相同，请检查是否误填！\n\n' +
                  '左侧应填SKU（如 JS-016-01），右侧应填数量（如 40）');
            return;
        }

        // [debug][2026-08-12] 检测数量列表是否全是数字
        const nonNumericQty = qtyArr.filter(v => !/^\d+$/.test(v));
        if (nonNumericQty.length > 0) {
            receiveDebug('fill-qty-not-numeric', {
                nonNumericCount: nonNumericQty.length,
                sample: nonNumericQty.slice(0, 5)
            });

            alert(`数量列表中有 ${nonNumericQty.length} 个非数字值，请检查！\n\n` +
                  `示例：${nonNumericQty.slice(0, 3).join(', ')}\n\n` +
                  '数量列表应该全是数字（如 40、110、20）');
            return;
        }

        const skuMap = {};

        skuArr.forEach((sku, index) => {
            skuMap[sku] = qtyArr[index];
        });

        const tableInfo = findReceiveTable();

        if (!tableInfo) {

            receiveDebug('fill-table-missing', {
                totalTables: document.querySelectorAll('table').length
            });

            alert("未找到表格");

            return;
        }

        // ========== 新增：虚拟列表滚动填表逻辑 ==========
        const virtualInfo = getReceiveVirtualInfo(tableInfo.table);
        const isVirtual = virtualInfo && virtualInfo.virtualRoot;

        receiveDebug('fill-virtual-check', {
            isVirtual,
            virtualInfo
        });

        let success = 0;
        const targetResults = skuArr.map((sku, index) => ({
            sku,
            qty: qtyArr[index],
            matchedRows: [],
            writes: []
        }));
        const writtenRefs = [];
        const filledSkus = new Set(); // 已填写的 SKU

        // 获取滚动容器
        let scrollContainer = null;
        if (isVirtual && virtualInfo.containers && virtualInfo.containers.length > 0) {
            // 取第一个可滚动的容器
            scrollContainer = virtualInfo.containers.find(c => {
                const elem = document.querySelector(`.${c.className.split(' ')[0]}`);
                return elem && elem.scrollHeight > elem.clientHeight;
            });
            if (scrollContainer) {
                const elem = document.querySelector(`.${scrollContainer.className.split(' ')[0]}`);
                scrollContainer = elem;
            }
        }

        // 如果没找到虚拟列表容器，尝试常规滚动容器
        if (!scrollContainer) {
            scrollContainer = tableInfo.table.closest('.bs_newtable-body')
                || tableInfo.table.closest('.bs-micro-modal-body')
                || tableInfo.table.closest('[style*="overflow"]');
        }

        receiveDebug('fill-scroll-container', {
            found: !!scrollContainer,
            tagName: scrollContainer?.tagName,
            className: scrollContainer?.className,
            scrollHeight: scrollContainer?.scrollHeight,
            clientHeight: scrollContainer?.clientHeight,
            scrollTop: scrollContainer?.scrollTop
        });

        const msgEl = document.getElementById('msg');
        msgEl.innerText = '开始填写...';

        let scrollAttempts = 0;
        const MAX_SCROLL_ATTEMPTS = 50;

        // 滚动填表循环
        while (scrollAttempts < MAX_SCROLL_ATTEMPTS && filledSkus.size < skuArr.length) {
            scrollAttempts++;

            receiveDebug('fill-scroll-attempt', {
                attempt: scrollAttempts,
                filled: filledSkus.size,
                target: skuArr.length
            });

            msgEl.innerText = `填写中 ${filledSkus.size}/${skuArr.length} (滚动第${scrollAttempts}轮)`;

            // 重新获取当前可见行
            const rows = Array.from(tableInfo.table.querySelectorAll('tbody > tr'));
            let filledThisRound = 0;

            receiveDebug('fill-round-scan', {
                attempt: scrollAttempts,
                visibleRows: rows.length
            });

            // 扫描并填写当前可见行
            rows.forEach((row, rowIndex) => {
                const sku = getRowSku(row, skuMap);

                // [debug][2026-08-12] 记录每行的匹配情况，帮助排查为什么某些行匹配不上
                if (scrollAttempts <= 2 || (sku && !(sku in skuMap)) || (sku && filledSkus.has(sku))) {
                    const rowText = (row.innerText || row.textContent || '').replace(/\s+/g, ' ').trim().slice(0, 120);
                    receiveDebug('fill-row-check', {
                        attempt: scrollAttempts,
                        rowIndex,
                        resolvedSku: sku,
                        inSkuMap: sku ? (sku in skuMap) : false,
                        alreadyFilled: sku ? filledSkus.has(sku) : false,
                        rowText
                    });
                }

                if (!sku || !(sku in skuMap) || filledSkus.has(sku)) {
                    return;
                }

                const input = findRowReceiveInput(row);

                if (!input) {
                    return;
                }

                const value = skuMap[sku];
                const beforeValue = input.value;

                // 填写
                input.focus();
                const nativeInputValueSetter = Object.getOwnPropertyDescriptor(
                    window.HTMLInputElement.prototype,
                    'value'
                ).set;
                nativeInputValueSetter.call(input, value);
                input.dispatchEvent(new Event('input', { bubbles: true }));
                input.dispatchEvent(new Event('change', { bubbles: true }));
                input.dispatchEvent(new KeyboardEvent('keyup', { bubbles: true }));
                input.blur();

                const afterValue = input.value;
                row.style.background = 'rgba(0,255,0,.15)';

                // 记录
                filledSkus.add(sku);
                filledThisRound++;

                const targetResult = targetResults.find(r => r.sku === sku);
                if (targetResult) {
                    targetResult.writes.push({
                        rowIndex,
                        beforeValue,
                        requestedValue: value,
                        afterValue,
                        inputConnected: input.isConnected,
                        rowConnected: row.isConnected
                    });
                }

                writtenRefs.push({
                    sku,
                    rowIndex,
                    requestedValue: value,
                    input,
                    row
                });

                receiveDebug('fill-row-written', {
                    sku,
                    rowIndex,
                    beforeValue,
                    requestedValue: value,
                    afterValue,
                    scrollAttempt: scrollAttempts
                });

                success++;
            });

            receiveDebug('fill-round-result', {
                attempt: scrollAttempts,
                filledThisRound,
                totalFilled: filledSkus.size,
                remaining: skuArr.length - filledSkus.size
            });

            // 全部填完，退出
            if (filledSkus.size >= skuArr.length) {
                break;
            }

            // 如果本轮没填到新的，需要滚动
            if (filledThisRound === 0 && scrollContainer) {
                const beforeScrollTop = scrollContainer.scrollTop;
                scrollContainer.scrollTop += 500;

                // 等待虚拟列表渲染
                await new Promise(resolve => setTimeout(resolve, 300));

                const afterScrollTop = scrollContainer.scrollTop;

                receiveDebug('fill-scroll-action', {
                    attempt: scrollAttempts,
                    beforeScrollTop,
                    afterScrollTop,
                    scrollDelta: afterScrollTop - beforeScrollTop,
                    scrollHeight: scrollContainer.scrollHeight,
                    clientHeight: scrollContainer.clientHeight,
                    reachedBottom: afterScrollTop >= scrollContainer.scrollHeight - scrollContainer.clientHeight - 10
                });

                // 滚动不动了（到底了）
                if (Math.abs(afterScrollTop - beforeScrollTop) < 10) {
                    receiveDebug('fill-scroll-stuck', {
                        message: '滚动到底，最后确认一轮'
                    });

                    // 等待一下，再扫最后一轮
                    await new Promise(resolve => setTimeout(resolve, 500));

                    const finalRows = Array.from(tableInfo.table.querySelectorAll('tbody > tr'));
                    finalRows.forEach(row => {
                        const sku = getRowSku(row, skuMap);
                        if (!sku || !(sku in skuMap) || filledSkus.has(sku)) return;

                        const input = findRowReceiveInput(row);
                        if (!input) return;

                        const value = skuMap[sku];
                        const beforeValue = input.value;

                        input.focus();
                        const nativeInputValueSetter = Object.getOwnPropertyDescriptor(
                            window.HTMLInputElement.prototype, 'value'
                        ).set;
                        nativeInputValueSetter.call(input, value);
                        input.dispatchEvent(new Event('input', { bubbles: true }));
                        input.dispatchEvent(new Event('change', { bubbles: true }));
                        input.dispatchEvent(new KeyboardEvent('keyup', { bubbles: true }));
                        input.blur();

                        const afterValue = input.value;
                        row.style.background = 'rgba(0,255,0,.15)';

                        filledSkus.add(sku);
                        success++;

                        receiveDebug('fill-row-written-final', {
                            sku,
                            beforeValue,
                            requestedValue: value,
                            afterValue
                        });
                    });

                    break; // 到底了，退出
                }
            } else if (filledThisRound === 0 && !scrollContainer) {
                // 没有滚动容器且本轮无新填写，退出
                receiveDebug('fill-no-scroll-container', {
                    message: '无滚动容器且当前行已扫完'
                });
                break;
            }
        }
        // ========== 虚拟列表滚动填表逻辑结束 ==========

        receiveDebug('fill-summary', {
            requestedCount: skuArr.length,
            success,
            scrollAttempts,
            missingTargets: targetResults
                .filter(result => result.writes.length === 0)
                .map(result => result.sku),
            matchedWithoutWrite: targetResults
                .filter(result =>
                    result.matchedRows.length > 0 &&
                    result.writes.length === 0
                )
                .map(result => result.sku),
            targetResults,
            virtual: getReceiveVirtualInfo(tableInfo.table)
        });

        [100, 1000].forEach(delay => {
            setTimeout(() => {
                const currentRows = Array.from(
                    tableInfo.table.querySelectorAll('tbody > tr')
                );

                receiveDebug(`fill-after-${delay}ms`, {
                    success,
                    written: writtenRefs.map(item => ({
                        sku: item.sku,
                        rowIndex: item.rowIndex,
                        requestedValue: item.requestedValue,
                        actualValue: item.input.value,
                        inputConnected: item.input.isConnected,
                        rowConnected: item.row.isConnected,
                        rowRect: getReceiveRectInfo(item.row)
                    })),
                    currentRows: currentRows.map((row, rowIndex) => {
                        const rowText = (
                            row.innerText || row.textContent || ''
                        ).replace(/\s+/g, ' ').trim();

                        return {
                            rowIndex,
                            dataRowIndex:
                                row.getAttribute('data-rowindex') || '',
                            connected: row.isConnected,
                            targetHits: skuArr.filter(
                                sku => rowTextContainsSku(rowText, sku)
                            ),
                            rect: getReceiveRectInfo(row),
                            rowText: rowText.slice(0, 180)
                        };
                    }),
                    virtual: getReceiveVirtualInfo(
                        tableInfo.table
                    )
                });
            }, delay);
        });

        const missingCount = skuArr.length - success;
        const missingSummary = missingCount > 0 ? `，未找到 ${missingCount} 个` : '';

        document.getElementById('msg')
            .innerText =
            `成功填写 ${success}/${skuArr.length} 个SKU${missingSummary}`;
    };

                            warehouseName: item.warehouseName
                        }))
                    })
                );

                if (json.code === 0 && json.data && json.data.page) {
                    json.data.page.rows.forEach(item => {
                        const responseSku = item.sku || item.skuName;
                        if (!responseSku) return;

                        skuDataMap[inventoryKey(group.warehouseId, responseSku)] = {

                        warehouseId: record.warehouse.id,
                        sku: record.sku

    // 自动收集订单号并批量获取ID（与SKU查询同时进行）
    async function autoFetchOrderIds() {
        if (!isInventoryPage()) return;

        const tdSel = '#pageList .list_items tbody tr td.order_No, #pageList .list_items tbody tr td:nth-child(5)';
        await waitForSelector(tdSel);

        const orderTds = document.querySelectorAll(tdSel);
        const orderNos = [];
        orderTds.forEach(td => {
            const text = td.innerText.replace(/\s+/g, '').trim();
            if (text) {
                orderNos.push(text);
            }
        });

        if (!orderNos.length) return;

        try {
            orderIdCache = {};
            packageWarehouseCache = {};

            const uniqueOrderNos = [...new Set(orderNos)];
            const orderBatches = [];
            for (let i = 0; i < uniqueOrderNos.length; i += BATCH_SIZE) {
                orderBatches.push(uniqueOrderNos.slice(i, i + BATCH_SIZE));
            }

            for (const batch of orderBatches) {
                const resp = await fetch('/api/v1/order/new/pageList.json', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                        'clienttype': '1'
                    },
                    body: JSON.stringify({
                        searchType: 'orderNo',
                        searchContent: batch.join(','),
                        inquireType: 2,
                        allOrder: true,
                        historyOrder: 0,
                        pageNo: 1,
                        pageSize: 300,
                        desc: 1,
                        orderBy: 'printTime'
                    })
                });

                const json = await resp.json();
                if (json.code !== 0 || !json.data?.page?.rows) continue;

                json.data.page.rows.forEach(order => {
                    if (order.platformOrderId) {
                        orderIdCache[order.platformOrderId] = order.id;
                    }
                    if (order.packageNo && order.warehouseId) {
                        packageWarehouseCache[String(order.packageNo).toUpperCase()] = {
                            id: String(order.warehouseId),
                            name: order.shipmentWarehouse || `仓库 ${order.warehouseId}`
                        };
                    }
                });
            }

            console.log(
                `[c仓助手] 已缓存 ${Object.keys(orderIdCache).length} 个订单ID，` +
                `${Object.keys(packageWarehouseCache).length} 个包裹仓库`
            );
            injectWarehouseTags();
        } catch (e) {
            console.log('[c仓助手] 获取订单ID失败:', e.message);
        }
    }

    // 点击单个包裹的按钮：用缓存的ID查该订单详情
    async function queryReplaceForPackage(btn, titleEl) {
        const origText = btn.innerText;
        btn.innerText = '查询中...';
        btn.disabled = true;

        try {
            // 找到该包裹对应的表格
            const listItemsOut = titleEl.closest('.list_items_out');
            if (!listItemsOut) return;

            // 获取订单号
            const orderTd = listItemsOut.querySelector(
                'table.list_items tbody tr td.order_No, table.list_items tbody tr td:nth-child(5)'
            );
            if (!orderTd) return;

            const orderNo = orderTd.innerText.trim();
            if (!orderNo || !/^\d+$/.test(orderNo)) {
                btn.innerText = '无订单号';
                setTimeout(() => { btn.innerText = origText; }, 2000);
                return;
            }

            // 从缓存获取订单ID，若无则单独查询
            let orderId = orderIdCache[orderNo];
            if (!orderId) {
                const resp1 = await fetch('/api/v1/order/new/pageList.json', {
                    method: 'POST',
                    headers: {
                        'Content-Type': 'application/json',
                        'clienttype': '1'
                    },
                    body: JSON.stringify({
                        searchType: 'orderNo',
                        searchContent: orderNo,
                        inquireType: 2,
                        allOrder: true,
                        historyOrder: 0,
                        pageNo: 1,
                        pageSize: 10,
                        desc: 1,
                        orderBy: 'printTime'
                    })
                });
                const json1 = await resp1.json();
                if (json1.code === 0 && json1.data && json1.data.page && json1.data.page.rows.length) {
                    orderId = json1.data.page.rows[0].id;
                    orderIdCache[orderNo] = orderId;
                }
            }

            if (!orderId) {
                btn.innerText = '未找到订单';
                setTimeout(() => { btn.innerText = origText; }, 2000);
                return;
            }

            // 查详情
            const resp2 = await fetch(
                `/api/v1/order/detail.json?id=${orderId}&historyOrder=0&viewBuyerMessage=false`,
                { headers: { 'clienttype': '1' } }
            );
            const json2 = await resp2.json();

            if (json2.code !== 0 || !json2.data) {
                btn.innerText = '查询失败';
                setTimeout(() => { btn.innerText = origText; }, 2000);
                return;
            }

            const detail = json2.data.orderDetail;
            const replaces = [];

            if (detail.orderItemVoList) {
                detail.orderItemVoList.forEach(item => {
                    if (item.inventorySku && item.inventorySku !== item.varSku) {
                        replaces.push({
                            from: item.varSku,
                            to: item.inventorySku
                        });
                    }
                });
            }

            if (replaces.length > 0) {
                btn.innerText = '有换货';
                btn.style.background = '#fff1f0';
                btn.style.borderColor = '#cf1322';
                btn.style.color = '#cf1322';

                // 在标题栏显示换货详情
                let tag = titleEl.querySelector('.c_replace_tag');
                if (!tag) {
                    tag = document.createElement('span');
                    tag.className = 'c_replace_tag';
                    tag.style.cssText =
                        'margin-left:8px;padding:2px 6px;border-radius:3px;font-size:11px;background:#fff1f0;color:#cf1322;';
                    titleEl.appendChild(tag);
                }
                tag.innerHTML = replaces.map(
                    r => `${r.from} → ${r.to}`
                ).join(', ');
            } else {
                btn.innerText = '无换货';
                btn.style.background = '#f6ffed';
                btn.style.borderColor = '#52c41a';
                btn.style.color = '#52c41a';
            }

        } catch (e) {
            btn.innerText = '出错';
            setTimeout(() => { btn.innerText = origText; }, 2000);
        } finally {
            btn.disabled = false;
        }
    }

    injectReplaceGoodsBtn();
    autoFetchOrderIds();

})();
