    function injectCostLabels() {

        skuCells.forEach(cell => {

                if (!container) return;
                const pTag = container.querySelector('p');

                const qtySpan = pTag.querySelector('.order_num');

            });

    // 拦截 XMLHttpRequest 获取订单列表中的仓库信息（页面用 axios/XHR）
    const origXhrOpen = XMLHttpRequest.prototype.open;
    const origXhrSend = XMLHttpRequest.prototype.send;
    XMLHttpRequest.prototype.open = function (method, url, ...rest) {
        this._url = url;
        return origXhrOpen.call(this, method, url, ...rest);
    };
    XMLHttpRequest.prototype.send = function (...args) {
        this.addEventListener('load', function () {
            try {
                if (this._url && this._url.includes('/api/v1/order/new/pageList.json')) {
                    const json = JSON.parse(this.responseText);
                    if (json.code === 0 && json.data?.page?.rows?.length) {
                        json.data.page.rows.forEach(order => {
                            const whName = order.shipmentWarehouse
                                || (warehouseList.find(w => w.id === String(order.warehouseId))?.name)
                                || '';
                            if (order.packageNo) {
                                if (whName) packageWarehouseMap[order.packageNo] = whName;
                                if (order.platformOrderId) packageOrderMap[order.packageNo] = order.platformOrderId;
                            }
                        });
                        setTimeout(injectOrderUI, 500);
                        setTimeout(injectCostLabels, 800);
                    }
                }
            } catch (e) {}
        });
        return origXhrSend.apply(this, args);
    };

    // GM_xmlhttpRequest 封装为 Promise
    function gmFetch(url, data) {
        return new Promise((resolve, reject) => {
            console.log('[Bigseller助手][gmFetch] 发送请求:', url, JSON.stringify(data).substring(0, 200));
            GM_xmlhttpRequest({
                method: 'POST',
                url,
                headers: { 'Content-Type': 'application/json' },
                data: JSON.stringify(data),
                onload: (resp) => {
                    console.log('[Bigseller助手][gmFetch] 响应状态:', resp.status, '内容:', resp.responseText.substring(0, 500));
                    try { resolve(JSON.parse(resp.responseText)); }
                    catch (e) { console.error('[Bigseller助手][gmFetch] JSON解析失败:', resp.responseText.substring(0, 200)); reject(e); }
                },
                onerror: (e) => { console.error('[Bigseller助手][gmFetch] 网络错误:', e); reject(e); }
            });
        });
    }

    // SKU 工具函数
    function extractStylePrefix(sku) {
        const parts = sku.split('-');
        if (parts.length >= 2) return parts.slice(0, 2).join('-');
        return sku;
    }
    function parseTitleInfo(title) {
        const parts = title.split(';');
        const pattern = parts.length >= 2 ? parts[parts.length - 2].trim() : '';
        const sizeMatch = title.match(/(\d+)\s*[\*xX×]\s*(\d+)/);
        const width = sizeMatch ? parseInt(sizeMatch[1]) : 0;
        const height = sizeMatch ? parseInt(sizeMatch[2]) : 0;
        return { pattern, width, height, area: width * height };
    }
    function sortRecommendations(list, currentTitle) {
        const current = parseTitleInfo(currentTitle);
        function getPriority(info) {
            const sameSize = info.width === current.width && info.height === current.height;
            const samePattern = info.pattern === current.pattern;
            const largerSize = info.area > current.area;
            if (sameSize && !samePattern) return { level: 1, label: '异图同尺' };
            if (samePattern && largerSize) return { level: 2, label: '同图更大' };
            if (!samePattern && largerSize) return { level: 3, label: '异图更大' };
            return { level: 4, label: '' };
        }
        return list.map(item => {
            const info = parseTitleInfo(item.title);
            const pri = getPriority(info);
            return { ...item, _priority: pri.level, _label: pri.label, _info: info };
        }).sort((a, b) => {
            const aHas = a.available > 0 ? 1 : 0, bHas = b.available > 0 ? 1 : 0;
            if (aHas !== bHas) return bHas - aHas;
            if (a._priority !== b._priority) return a._priority - b._priority;
            if (a._priority === 2 || a._priority === 3) {
                const aSH = a._info.height === current.height ? 0 : 1;
                const bSH = b._info.height === current.height ? 0 : 1;
                if (aSH !== bSH) return aSH - bSH;
                if (a._info.area !== b._info.area) return a._info.area - b._info.area;
            }
            return (b.available ?? 0) - (a.available ?? 0);
        });
    }

        // 确认换货
        document.getElementById('c_swap_confirm').onclick = async () => {
            const rows = mask.querySelectorAll('.c_sku_row');
            const tasks = [];
            rows.forEach(row => {
                const targetSku = row.querySelector('.c_target_input').value.trim();
                const originalSku = row.dataset.sku;
                const rootSku = row.dataset.rootSku || originalSku;
                const batchChecked = row.querySelector('.c_batch_check')?.checked || false;
                // 有目标SKU就加入任务；没有目标SKU但换了仓库也加入（纯换仓）

                    tasks.push({ originalSku, rootSku, targetSku: originalSku, orderItemId: row.dataset.orderItemId, batchChecked });

            if (!tasks.length) { alert('请至少为一个SKU指定目标'); return; }

            const confirmBtn = document.getElementById('c_swap_confirm');
            confirmBtn.innerText = '换货中...';
            confirmBtn.disabled = true;
            let success = 0, fail = 0;
            const successItems = []; // 收集成功的换货项，最后统一保存

            for (let i = 0; i < tasks.length; i++) {
                const t = tasks[i];
                confirmBtn.innerText = `换货中 ${i + 1}/${tasks.length}...`;
                console.log('[Bigseller助手][换货][DEBUG] 任务%d:', i, JSON.stringify({ orderNo, orderId, originalSku: t.originalSku, rootSku: t.rootSku, targetSku: t.targetSku, orderItemId: t.orderItemId }));
                try {
                    const selectedWhId = getWarehouseId();

                    const isSameWarehouse = selectedWhId === orderWarehouseId;

                    if (isSameSku && isSameWarehouse) {
                        console.log('[Bigseller助手][换货] 同仓同SKU，跳过'); continue;

                        if (!whOk) { console.log('[Bigseller助手][换货] 换仓失败'); fail++; continue; }
                        console.log('[Bigseller助手][换货] 换仓成功');
                    }

                    // 步骤2：如果需要换SKU，调 changeMappingRelation
                    if (!isSameSku) {
                        const targetInfo = await fetchSkuInfo(t.targetSku);
                        if (!targetInfo || !targetInfo.skuId) { console.log('[Bigseller助手][换货][DEBUG] 目标SKU未找到:', t.targetSku); fail++; continue; }
                        console.log('[Bigseller助手][换货] 步骤2: 换SKU %s → %s (skuId=%s)', t.originalSku, t.targetSku, targetInfo.skuId);
                        swapOk = await doSwapCall(orderId, t.orderItemId, selectedWhId, targetInfo.skuId, t.targetSku);
                        console.log('[Bigseller助手][换货][DEBUG] doSwapCall结果: swapOk=%s', swapOk);
                        if (swapOk) {
                            successItems.push({ rootSku: t.rootSku, originalSku: t.originalSku, targetSku: t.targetSku, targetSkuId: String(targetInfo.skuId), targetTitle: targetInfo.title, targetImage: targetInfo.image, targetAvailable: targetInfo.available });
                            const skuSpan = findSkuSpanInOrder(orderNo, t.rootSku) || findSkuSpanInOrder(orderNo, t.originalSku);
                            if (skuSpan) injectSwapResult(skuSpan, t.targetSku, targetInfo, t.rootSku, orderWarehouseId);
                        }
                    } else {

                        swapOk = true;
                        const whName = warehouseList.find(w => w.id === selectedWhId)?.name || selectedWhId;
                        const origInfo = await fetchSkuInfo(t.originalSku);
                        successItems.push({ rootSku: t.rootSku, originalSku: t.originalSku, targetSku: t.originalSku, targetSkuId: origInfo?.skuId ? String(origInfo.skuId) : '', targetTitle: origInfo?.title || '', targetImage: origInfo?.image || '', targetAvailable: origInfo?.available || '' });
                        const skuSpan = findSkuSpanInOrder(orderNo, t.rootSku) || findSkuSpanInOrder(orderNo, t.originalSku);
                        if (skuSpan) {
                            injectSwapResult(skuSpan, t.originalSku, {
                                title: `${origInfo?.title || ''} [${whName}]`,
                                available: origInfo?.available ?? '--',
                                image: origInfo?.image || ''
                            }, t.rootSku, orderWarehouseId);
                        }
                    }

                    if (swapOk) {
                        success++;
                        // 批量应用到其他同SKU订单
                        if (t.batchChecked && sameSkuOrdersMap[t.originalSku]?.length) {
                            const batchOrders = sameSkuOrdersMap[t.originalSku];
                            const skippedMultiSku = []; // 记录跳过的多SKU订单
                            for (let j = 0; j < batchOrders.length; j++) {
                                try {
                                    const bo = batchOrders[j];
                                    const dResp = await fetch(`/api/v1/order/detail.json?id=${bo.id}&historyOrder=0&viewBuyerMessage=false`, { headers: { 'clienttype': '1' } });
                                    const dJson = await dResp.json();
                                    if (dJson.code !== 0 || !dJson.data?.orderDetail?.orderItemVoList?.length) continue;
                                    const itemList = dJson.data.orderDetail.orderItemVoList;
                                    // 多SKU订单跳过，避免换仓导致其他SKU异仓
                                    if (itemList.length > 1 && selectedWhId !== String(bo.warehouseId || orderWarehouseId)) {
                                        skippedMultiSku.push(bo.platformOrderId);
                                        continue;
                                    }
                                    const oItem = itemList.find(item => item.varSku === t.originalSku || item.inventorySku === t.originalSku);
                                    if (!oItem) continue;
                                    const boWhId = String(bo.warehouseId || orderWarehouseId);
                                    let bOk = false;
                                    if (selectedWhId !== boWhId) {
                                        const bWhOk = await doChangeWarehouse(bo.id, selectedWhId);
                                        if (!bWhOk) continue;
                                    }
                                    if (!isSameSku) {
                                        const bTargetInfo = await fetchSkuInfo(t.targetSku);
                                        if (bTargetInfo?.skuId) {
                                            bOk = await doSwapCall(bo.id, oItem.id, selectedWhId, bTargetInfo.skuId, t.targetSku);
                                        }
                                    } else {
                                        bOk = true;
                                    }
                                    if (bOk) {
                                        const bItem = { rootSku: oItem.varSku || t.originalSku, originalSku: t.originalSku, targetSku: isSameSku ? t.originalSku : t.targetSku, targetSkuId: '', targetTitle: '', targetImage: '', targetAvailable: '' };
                                        saveSwapRecord({ orderNo: bo.platformOrderId, orderId: String(bo.id), rootWarehouseId: boWhId, items: [bItem] });
                                    }
                                } catch (e) {}
                                confirmBtn.innerText = `批量 ${t.originalSku} ${j + 1}/${batchOrders.length}...`;
                            }
                            if (skippedMultiSku.length) {
                                alert(`以下多SKU订单已跳过（需换仓但可能导致其他SKU异仓），请手动处理：\n${skippedMultiSku.join('\n')}`);
                            }
                        }
                    }
                } catch (e) { fail++; console.log('[Bigseller助手][换货] 异常:', e.message); }
            }

            // 统一保存本订单的换货记录
            if (successItems.length) {
                const saveResult = await saveSwapRecord({ orderNo, orderId: String(orderId), rootWarehouseId: orderWarehouseId, items: successItems });
                console.log('[Bigseller助手][换货][DEBUG] 统一保存结果:', JSON.stringify(saveResult));
            }

            alert(`换货完成！成功: ${success}，失败: ${fail}`);
            // 刷新仓库标签显示
            if (success > 0) {
                const selectedWhId = getWarehouseId();
                const whName = warehouseList.find(w => w.id === selectedWhId)?.name || selectedWhId;
                // 找到当前订单的包裹号并更新 packageWarehouseMap
                Object.keys(packageOrderMap).forEach(pkgNo => {
                    if (packageOrderMap[pkgNo] === orderNo) {
                        packageWarehouseMap[pkgNo] = whName;
                        // 更新页面上的仓库标签
                        const tag = document.querySelector(`.c_wh_tag`);
                        document.querySelectorAll('.c_wh_tag').forEach(el => {
                            const span = el.previousSibling;
                            if (span && span.innerText && span.innerText.trim().replace(/^#\s*/, '') === pkgNo) {
                                el.innerText = whName;
                            }
                        });
                    }
                });
            }
            closeModal();
        };
    }

    // 取消换货核心逻辑
    async function doCancelSwap(orderNo, originalSku, targetSku, rootSku, rootWarehouseId) {
        try {
            console.log('[Bigseller助手][取消换货][DEBUG] 开始:', JSON.stringify({ orderNo, originalSku, targetSku, rootSku, rootWarehouseId }));
            // 查询订单当前状态
            const orderResp = await fetch('/api/v1/order/new/pageList.json', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json', 'clienttype': '1' },
                body: JSON.stringify({ searchType: 'orderNo', searchContent: orderNo, inquireType: 2, allOrder: true, historyOrder: 0, pageNo: 1, pageSize: 10, desc: 1, orderBy: 'printTime' })
            });
            const orderJson = await orderResp.json();
            if (orderJson.code !== 0 || !orderJson.data?.page?.rows?.length) return { success: false, msg: '未找到订单' };
            const cancelOrderRow = orderJson.data.page.rows[0];
            const orderId = cancelOrderRow.id;
            const currentWhId = String(cancelOrderRow.warehouseId);
            const needChangeWh = rootWarehouseId && rootWarehouseId !== currentWhId;
            const needChangeSku = rootSku !== targetSku;
            console.log('[Bigseller助手][取消换货][DEBUG] orderId=%s, currentWhId=%s, rootWarehouseId=%s, needChangeWh=%s, needChangeSku=%s, 恢复目标: rootSku=%s', orderId, currentWhId, rootWarehouseId, needChangeWh, needChangeSku, rootSku);

            // 步骤1：如果需要换仓，先换回原仓库
            if (needChangeWh) {
                const whOk = await doChangeWarehouse(orderId, rootWarehouseId);
                if (!whOk) return { success: false, msg: '换回原仓库失败' };
                console.log('[Bigseller助手][取消换货][DEBUG] 换仓成功 → %s', rootWarehouseId);
            }

            // 步骤2：如果需要换SKU，换回rootSku（平台原始SKU）
            if (needChangeSku) {
                const restoreWhId = rootWarehouseId || currentWhId;
                // 在恢复目标仓库中查rootSku信息
                const rootSkuInfo = await fetchSkuInfo(rootSku, restoreWhId);
                console.log('[Bigseller助手][取消换货][DEBUG] fetchSkuInfo(%s, wh=%s) 结果:', rootSku, restoreWhId, rootSkuInfo ? JSON.stringify({ skuId: rootSkuInfo.skuId }) : 'null');
                if (!rootSkuInfo || !rootSkuInfo.skuId) return { success: false, msg: `未找到根SKU信息: ${rootSku} (仓库: ${restoreWhId})` };
                // 获取订单详情找到对应orderItemId
                const detailResp = await fetch(`/api/v1/order/detail.json?id=${orderId}&historyOrder=0&viewBuyerMessage=false`, { headers: { 'clienttype': '1' } });
                const detailJson = await detailResp.json();
                if (detailJson.code !== 0 || !detailJson.data?.orderDetail?.orderItemVoList?.length) return { success: false, msg: '获取订单详情失败' };
                const allItems = detailJson.data.orderDetail.orderItemVoList;
                // 用rootSku(varSku)匹配商品项
                const orderItem = allItems.find(item => item.varSku === rootSku);
                if (!orderItem) return { success: false, msg: `未找到商品项: ${rootSku}` };
                const swapOk = await doSwapCall(orderId, orderItem.id, restoreWhId, rootSkuInfo.skuId, rootSku);
                if (!swapOk) return { success: false, msg: '换回原SKU失败' };
                console.log('[Bigseller助手][取消换货][DEBUG] 换SKU成功 → %s', rootSku);
            }

            return { success: true, msg: '成功' };
        } catch (e) { return { success: false, msg: e.message }; }
    }

    }

    async function cancelOrderSwap(orderNo) {

        const record = records[0]; // 新结构：一订单一条记录

        if (!confirm(`确认取消该订单的全部 ${items.length} 条换货？将恢复原SKU映射`)) return;
        let success = 0; const failures = [];
        for (const item of items) {
            const result = await doCancelSwap(orderNo, item.originalSku, item.targetSku, item.rootSku, record.rootWarehouseId);
            if (result.success) success++;
            else failures.push(`${item.rootSku}: ${result.msg}`);
        }
        // 成功后删除整条记录
        if (success > 0) {
            await deleteSwapRecord(orderNo);
        }
        // 移除页面上该订单的换货样式，恢复原SKU显示
        const odnSpans = document.querySelectorAll('td.odn_el div.copy_txt_box > div > span');

                const row = sp.closest('tr');

                row.querySelectorAll('.c_swap_styled').forEach(container => {
                    // 恢复SKU文本
                    const skuSpan = container.querySelector('span.f_default');
                    if (skuSpan && skuSpan.dataset.rootSku) {
                        skuSpan.innerText = skuSpan.dataset.rootSku;
                        delete skuSpan.dataset.rootSku;
                    }
                    // 恢复商品图片
                    const img = container.querySelector('.img_box_out img');
                    if (img && img.dataset.rootImage) {
                        img.src = img.dataset.rootImage;
                        delete img.dataset.rootImage;
                    }
                    // 恢复变体描述
                    const preInit = container.querySelector('span.pre_init');
                    if (preInit && preInit.dataset.rootTitle) {
                        preInit.innerText = preInit.dataset.rootTitle;
                        delete preInit.dataset.rootTitle;
                    }
                    // 移除标签和?图标
                    container.querySelectorAll('.c_swap_tag, .c_swap_info').forEach(el => el.remove());
                    // 恢复复制按钮
                    const copyBtn = container.querySelector('.copy_txt_box a.copy_btn');
                    if (copyBtn) copyBtn.style.display = '';
                    // 移除容器样式
                    container.classList.remove('c_swap_styled');
                    container.style.background = '';
                    container.style.border = '';
                    container.style.borderRadius = '';
                    container.style.padding = '';
                });
                // 移除"取消换货"按钮
                const pkgNo = Object.keys(packageOrderMap).find(k => packageOrderMap[k] === orderNo);
                if (pkgNo) {
                    document.querySelectorAll('.c_order_cancel_swap_btn').forEach(btn => {
                        const span = btn.parentElement?.querySelector('span');
                        if (span && span.innerText.trim().replace(/^#\s*/, '') === pkgNo) btn.remove();
                    });
                }
            }
        }
        // 更新仓库标签（恢复到rootWarehouseId对应的仓库名）
        if (success > 0 && record.rootWarehouseId) {
            const rootWhId = record.rootWarehouseId;
            const rootWhName = warehouseList.find(w => w.id === rootWhId)?.name || rootWhId;
            Object.keys(packageOrderMap).forEach(pkgNo => {
                if (packageOrderMap[pkgNo] === orderNo) {
                    packageWarehouseMap[pkgNo] = rootWhName;
                    document.querySelectorAll('.c_wh_tag').forEach(el => {
                        const span = el.previousSibling;
                        if (span && span.innerText && span.innerText.trim().replace(/^#\s*/, '') === pkgNo) {
                            el.innerText = rootWhName;
                        }
                    });
                }
            });
        }
        if (failures.length) {
            alert(`取消完成！成功: ${success}，失败: ${failures.length}\n\n失败详情:\n${failures.join('\n')}`);
        } else {
            alert(`取消完成！全部 ${success} 条成功`);
        }
    }

    async function batchCancelSwap() {
        const rowCheckboxes = document.querySelectorAll('tr td input[type=checkbox]:checked');
        const checkedOrderNos = [];
        rowCheckboxes.forEach(cb => {
            const row = cb.closest('tr');
            if (!row) return;

            if (oSpan) { const no = oSpan.innerText.trim(); if (no && /^\d+$/.test(no)) checkedOrderNos.push(no); }
        });
        if (!checkedOrderNos.length) { alert('未勾选任何订单'); return; }

        if (!records.length) { alert('勾选的订单中没有换货记录'); return; }
        // 新结构：每条记录有 items 数组，展开为逐项处理
        const allTasks = [];
        records.forEach(r => { (r.items || []).forEach(item => allTasks.push({ orderNo: r.orderNo, rootWarehouseId: r.rootWarehouseId, ...item })); });
        if (!allTasks.length) { alert('没有换货项'); return; }
        if (!confirm(`确认批量取消换货？将对 ${allTasks.length} 条换货恢复原SKU映射`)) return;

            const t = allTasks[i];

            const result = await doCancelSwap(t.orderNo, t.originalSku, t.targetSku, t.rootSku, t.rootWarehouseId);
            if (result.success) { success++; }
            else { fail++; }
        }
        // 删除所有相关记录并恢复原SKU显示
        const orderNosSet = new Set(records.map(r => r.orderNo));
        for (const oNo of orderNosSet) { await deleteSwapRecord(oNo); }
        document.querySelectorAll('tr').forEach(row => {
            const oSpan = row.querySelector('td.odn_el div.copy_txt_box > div > span');

                    const img = container.querySelector('.img_box_out img');
                    if (img && img.dataset.rootImage) {
                        img.src = img.dataset.rootImage;
                        delete img.dataset.rootImage;
                    }
                    const preInit = container.querySelector('span.pre_init');
                    if (preInit && preInit.dataset.rootTitle) {
                        preInit.innerText = preInit.dataset.rootTitle;
                        delete preInit.dataset.rootTitle;
                    }
                    container.querySelectorAll('.c_swap_tag, .c_swap_info').forEach(el => el.remove());
                    container.classList.remove('c_swap_styled');
                    container.style.background = '';
                    container.style.border = '';
                    container.style.borderRadius = '';
                    container.style.padding = '';
                });
            }
        });
        btn.innerText = originalText; btn.disabled = false;
        alert(`批量取消完成！成功: ${success}，失败: ${fail}`);
    }

        function getShelfSearchPairs(skuList, shelfList, pageWindow = getPageWindow()) {
            const rows = Array.from(pageWindow.document.querySelectorAll('table.list_items > tbody > tr.list_item'));
            const usedRows = new Set();
            const available = [];
            const unchanged = [];
            const missing = [];

            skuList.forEach((sku, index) => {
                const normalizedSku = normalizeShelfText(sku);
                const rowIndex = rows.findIndex((row, candidateIndex) => {
                    if (usedRows.has(candidateIndex)) return false;
                    return getInventoryRowSkuCandidates(row).includes(normalizedSku);
                });
                const pair = { sku, shelf: shelfList[index] };
                if (rowIndex < 0) {
                    missing.push(pair);
                    return;
                }
                usedRows.add(rowIndex);
                pair.currentShelf = getInventoryRowShelf(rows[rowIndex]);
                if (pair.currentShelf && pair.currentShelf === normalizeShelfText(pair.shelf)) {
                    unchanged.push(pair);
                } else {
                    available.push(pair);
                }
            });

            return { available, unchanged, missing, rowCount: rows.length };
        }

        function getShelfJobFingerprint(skuList, shelfList) {
            return JSON.stringify({ skuList, shelfList });
        }

        function readShelfJob() {
            try {
                const raw = localStorage.getItem(SHELF_JOB_STORAGE_KEY);
                if (raw) return JSON.parse(raw);

                // v1 的 300 条批次可能只提交了当前页 50 条，迁移时必须从头开始。
                const legacyRaw = localStorage.getItem(SHELF_LEGACY_JOB_STORAGE_KEY);
                if (!legacyRaw) return null;
                const legacyJob = JSON.parse(legacyRaw);
                return {
                    ...legacyJob,
                    version: 2,
                    batchSize: SHELF_BATCH_SIZE,
                    completedCount: 0,
                    missingItems: [],
                    failedItems: []
                };
            } catch (error) {
                return null;
            }
        }

        function writeShelfJob(job) {
            try {
                localStorage.setItem(SHELF_JOB_STORAGE_KEY, JSON.stringify({
                    ...job,
                    updatedAt: new Date().toISOString()
                }));
                return true;
            } catch (error) {
                return false;
            }
        }

        function clearShelfJob() {
            try {
                localStorage.removeItem(SHELF_JOB_STORAGE_KEY);
                localStorage.removeItem(SHELF_LEGACY_JOB_STORAGE_KEY);
            } catch (error) { /* storage unavailable */ }
        }

        function getOrCreateShelfJob(skuList, shelfList) {
            const fingerprint = getShelfJobFingerprint(skuList, shelfList);
            const saved = readShelfJob();
            if (saved?.fingerprint === fingerprint
                && Array.isArray(saved.skuList)
                && Array.isArray(saved.shelfList)
                && saved.skuList.length === skuList.length
                && saved.shelfList.length === shelfList.length
                && Number.isInteger(saved.completedCount)
                && saved.completedCount >= 0
                && saved.completedCount <= skuList.length) {
                saved.missingItems ||= [];
                saved.failedItems ||= [];
                return saved;
            }
            return {
                version: 2,
                fingerprint,
                skuList: [...skuList],
                shelfList: [...shelfList],
                batchSize: SHELF_BATCH_SIZE,
                completedCount: 0,
                missingItems: [],
                failedItems: [],
                updatedAt: new Date().toISOString()
            };
        }

        function splitShelfBatches(skuList, shelfList, startIndex = 0) {
            const batches = [];
            for (let start = startIndex; start < skuList.length; start += SHELF_BATCH_SIZE) {
                const end = Math.min(start + SHELF_BATCH_SIZE, skuList.length);
                batches.push({
                    start,
                    end,
                    skuList: skuList.slice(start, end),
                    shelfList: shelfList.slice(start, end)
                });
            }
            return batches;
        }

        async function waitForShelfModalConfirmation(timeout = 30 * 60 * 1000) {
            const pageWindow = getPageWindow();
            const modal = await waitForElement(() => findVisibleElement('.ant-modal-wrap', pageWindow), 2000, 50);
            if (!modal) return { confirmed: false, action: 'missing-modal' };

            let action = 'pending';
            const onClick = event => {
                const button = event.target?.closest?.('button, .ant-btn');
                const text = button?.textContent?.replace(/\s+/g, '').trim() || '';
                if (text.includes('确定')) action = 'confirm';
                else if (text.includes('取消')) action = 'cancel';
            };
            modal.addEventListener('click', onClick, true);

            return new Promise(resolve => {
                const startedAt = Date.now();
                const finish = result => {
                    modal.removeEventListener('click', onClick, true);
                    resolve(result);
                };
                const check = () => {
                    const stillOpen = getModalRows(pageWindow).length > 0;
                    if (!stillOpen) {
                        finish({ confirmed: action === 'confirm', action });
                        return;
                    }
                    if (Date.now() - startedAt >= timeout) {
                        finish({ confirmed: false, action: 'timeout' });
                        return;
                    }
                    setTimeout(check, 250);
                };
                check();
            });
        }

            return Array.from(pageWindow.document.querySelectorAll('.ant-modal-wrap'))

                    const failureCount = modal.querySelector(resultFailureCountSelector);
                    return !!progress
                        && progressWidth === '100%'
                        && (!!successCount || !!failureCount
                            || /操作成功\s*[：:]\s*\d+/.test(resultContent.textContent || '')

        function findShelfOperationErrorMessage(pageWindow = getPageWindow()) {
            const errorPattern = /(锁定库存|拣货失败|货架位[^。\n]{0,40}失败|设置货架[^。\n]{0,40}失败)/i;

                    const errorMessage = findShelfOperationErrorMessage(pageWindow);
                    if (resultModal || errorMessage || Date.now() - startedAt >= timeout) {

                check();
            });
        }

        function parseShelfOperationResult(resultModal) {
            const resultContent = resultModal?.querySelector('#exportDetailContent');
            const progressElement = resultModal?.querySelector(resultProgressSelector);
            const progressWidth = progressElement?.style.width || null;
            const text = resultContent?.textContent || resultModal?.textContent || '';
            // 这两个是用户在结果弹窗中确认的精确路径，必须从弹窗根节点查找，
            // 不能从 #exportDetailContent 内部再次以 #exportDetailContent 开头查询。
            const successText = resultModal?.querySelector(resultSuccessCountSelector)?.textContent.trim();
            const failureText = resultModal?.querySelector(resultFailureCountSelector)?.textContent.trim();
            const success = Number(successText || (text.match(/操作成功\s*[：:]\s*(\d+)/) || [])[1]);
            const failure = Number(failureText || (text.match(/操作失败\s*[：:]\s*(\d+)/) || [])[1]);
            const detailLines = Array.from(resultModal?.querySelector(resultDetailSelector)?.children || [])
                .map(item => item.textContent.trim())
                .filter(Boolean);
            const reportedFailureSkuCandidates = detailLines.map(line => (
                line.match(/^(.+?)\s*:\s*(?:failed|失败|error)/i)?.[1]?.trim() || null
            )).filter(Boolean);
            return {
                success: Number.isFinite(success) ? success : null,
                failure: Number.isFinite(failure) ? failure : null,
                detailLines,
                reportedFailureSkuCandidates,
                progressValue: progressElement?.closest('[role="progressbar"]')?.getAttribute('aria-valuenow') || null,
                progressWidth,
                progressPathFound: !!progressElement,
                progressCompleted: !!resultContent?.querySelector('.ant-progress-status-success') && progressWidth === '100%'
            };
        }

                resultFound: !!resultModal,

                confirmDisabled: !!confirmButton.disabled,
                state: collectShelfModalLifecycleState(pageWindow)
            });

                warehouseSearchDebug && console.log('[debug][2026-07-23][货架提交错误提示已跳过]', {

                    errorMessage: operationOutcome.errorMessage,

                    action: 'toast-failure-skipped',

                    resultCountMatches: true,
                    detailLines: [operationOutcome.errorMessage],
                    reportedFailureSkuCandidates: (operationOutcome.errorMessage.match(/[A-Z]{2,}[\w-]*/g) || [])
                };
            }
            if (!resultModal) {
                warehouseSearchDebug && console.log('[debug][2026-07-23][货架结果弹窗等待超时]', {
                    expectedProcessedCount,
                    state: collectShelfModalLifecycleState(pageWindow)

                state: collectShelfModalLifecycleState(pageWindow)
            });

            const result = parseShelfOperationResult(resultModal);
            const reportedProcessedCount = Number.isFinite(result.success) && Number.isFinite(result.failure)

                : null;

                reportedSuccess: result.success,

            });

            // 防止页面只处理了部分 SKU 后脚本误进入下一批。
            if (!resultCountMatches) {
                return {
                    confirmed: false,
                    action: 'result-count-mismatch',
                    expectedProcessedCount,
                    reportedProcessedCount,
                    ...result
                };
            }
            const closeButton = findVisibleModalButton(resultModal, '关闭', pageWindow)
                || resultModal.querySelector('.ant-modal-footer > div > button');
            if (!closeButton) return { confirmed: false, action: 'missing-result-close', ...result };
            warehouseSearchDebug && console.log('[debug][2026-07-23][货架结果关闭]', {
                expectedProcessedCount,
                reportedProcessedCount,
                resultCountMatches,
                closeButtonPath: getElementCssPath(closeButton, pageDocument)
            });
            const resultCloseClick = clickPageElement(closeButton);
            warehouseSearchDebug && console.log('[debug][2026-07-23][货架结果关闭点击返回]', {
                resultCloseClick,
                state: collectShelfModalLifecycleState(pageWindow)
            });
            let resultClosed = await waitForElement(() => (
                !resultModal.isConnected || !isVisibleElement(resultModal, pageWindow) ? pageDocument.body : null
            ), 3000, 50);
            warehouseSearchDebug && console.log('[debug][2026-07-23][货架结果关闭等待]', {
                resultClosed: !!resultClosed,
                state: collectShelfModalLifecycleState(pageWindow)
            });
            if (!resultClosed) {
                const resultCloseFallback = dispatchMouseClick(closeButton, pageWindow);
                resultClosed = await waitForElement(() => (
                    !resultModal.isConnected || !isVisibleElement(resultModal, pageWindow) ? pageDocument.body : null
                ), 3000, 50);
                warehouseSearchDebug && console.log('[debug][2026-07-23][货架结果关闭鼠标兜底]', {
                    resultCloseFallback,
                    resultClosed: !!resultClosed,
                    state: collectShelfModalLifecycleState(pageWindow)
                });
            }
            if (!resultClosed) return { confirmed: false, action: 'result-close-timeout', ...result };

            let parentClosed = await waitForElement(() => (
                !modal.isConnected || !isVisibleElement(modal, pageWindow) ? pageDocument.body : null
            ), 3000, 50);
            if (!parentClosed) {
                const parentCloseButton = modal.querySelector(
                    '.ant-modal-close, .ant-modal-header .close, .ant-modal-content > button'
                );
                if (parentCloseButton) {
                    const parentCloseClick = clickPageElement(parentCloseButton);
                    parentClosed = await waitForElement(() => (
                        !modal.isConnected || !isVisibleElement(modal, pageWindow) ? pageDocument.body : null
                    ), 3000, 50);
                    warehouseSearchDebug && console.log('[debug][2026-07-23][货架设置弹窗关闭]', {
                        parentCloseClick,
                        parentClosed: !!parentClosed,
                        state: collectShelfModalLifecycleState(pageWindow)
                    });
                }
            }
            return parentClosed
                ? {
                    confirmed: true,
                    action: 'result-count-confirmed',
                    expectedProcessedCount,
                    reportedProcessedCount,
                    resultCountMatches,
                    ...result
                }
                : { confirmed: false, action: 'parent-close-timeout', ...result };
        }

        function getElementCssPath(element, pageDocument = getPageWindow().document) {
            if (!element || element.nodeType !== 1) return null;
            const parts = [];
            let current = element;
            while (current && current !== pageDocument.body && current.nodeType === 1) {
                let part = current.tagName.toLowerCase();
                if (current.id) {
                    part += `#${current.id}`;
                    parts.unshift(part);
                    break;
                }
                const classes = Array.from(current.classList || [])
                    .filter(className => /^[A-Za-z0-9_-]+$/.test(className))
                    .slice(0, 3);
                if (classes.length) part += `.${classes.join('.')}`;
                const parent = current.parentElement;
                if (parent) {
                    const sameTagSiblings = Array.from(parent.children)
                        .filter(child => child.tagName === current.tagName);
                    if (sameTagSiblings.length > 1) {
                        part += `:nth-of-type(${sameTagSiblings.indexOf(current) + 1})`;
                    }
                }
                parts.unshift(part);
                current = parent;
            }
            return parts.length ? `body > ${parts.join(' > ')}` : 'body';
        }

                setTimeout(() => logSnapshot('click-after-100ms', event.target), 100);
            }, true);

            pageDocument.addEventListener('focusin', event => {
                const target = event.target;
                if (!findVisibleElement('.ant-modal-wrap', pageWindow)) return;
                if (!/INPUT|TEXTAREA/.test(target.tagName) && !target.isContentEditable) return;
                logSnapshot('focusin', target);
            }, true);

            const observer = new pageWindow.MutationObserver(mutations => {
                const added = [];
                mutations.forEach(mutation => mutation.addedNodes.forEach(node => {
                    if (node.nodeType !== 1) return;
                    const text = node.textContent?.trim() || '';
                    const className = String(node.className || '');
                    if (/dropdown|option|item|select/i.test(className) || /下拉|货架/.test(text)) {
                        added.push({
                            path: getElementCssPath(node, pageDocument),
                            tag: node.tagName,
                            className,
                            text: text.slice(0, 240),
                            html: node.outerHTML.slice(0, 800)
                        });
                    }
                }));
                if (added.length) {
                    warehouseDebug && console.log('[debug][2026-07-23][货架输入监听][mutation]', JSON.stringify({
                        added: added.slice(0, 30),
                        activePath: getElementCssPath(pageDocument.activeElement, pageDocument)
                    }));
                }
            });
            observer.observe(pageDocument.body, { childList: true, subtree: true, attributes: true, attributeFilter: ['class', 'style', 'aria-expanded'] });
        }

        function getModalRowSku(row) {
            return row.querySelector('td.sku_el > div > div.ml_10 > div:nth-child(2) > div > div > span')?.textContent.trim() || '';
        }

        function normalizeShelfText(text) {
            return String(text || '').replace(/\s+/g, '').trim();
        }

        function getVisibleShelfCandidates(shelf, pageWindow = getPageWindow()) {
            const normalizedShelf = normalizeShelfText(shelf);
            const selectors = [
                '.ant-select-dropdown .option',
                '.ant-select-dropdown [role="option"]',
                '.ant-select-dropdown li',
                '.ant-select-dropdown-menu-item',
                '.ant-dropdown-menu-item',
                '.bs-antd-select-dropdown li',
                '[class*="dropdown"] [class*="option"]',
                '[class*="dropdown"] [class*="item"]'
            ];
            const selectorCandidates = selectors.flatMap(selector => Array.from(
                pageWindow.document.querySelectorAll(selector)
            ));
            return selectorCandidates.filter((option, index) => selectorCandidates.indexOf(option) === index)
                .filter(option => isVisibleElement(option, pageWindow))
                .filter(option => normalizeShelfText(option.textContent) === normalizedShelf);
        }

        function findVisibleShelfOption(shelf, pageWindow = getPageWindow()) {
            return getVisibleShelfCandidates(shelf, pageWindow)[0] || null;
        }

        function describeVisibleShelfNodes(shelf, pageWindow = getPageWindow()) {
            const normalizedShelf = normalizeShelfText(shelf);
            return Array.from(pageWindow.document.querySelectorAll('body *'))
                .filter(element => isVisibleElement(element, pageWindow))
                .filter(element => {
                    const text = normalizeShelfText(element.textContent);
                    return text && (text === normalizedShelf || text.includes(normalizedShelf));
                })
                .slice(0, 40)
                .map(element => ({
                    tag: element.tagName,
                    className: element.className,
                    role: element.getAttribute('role'),
                    text: element.textContent.trim().slice(0, 160),
                    childCount: element.children.length
                }));
        }

        function waitForShelfOption(shelf, pageWindow = getPageWindow(), timeout = 5000) {
            return new Promise(resolve => {
                const pageDocument = pageWindow.document;
                let completed = false;
                let observer = null;
                let intervalId = null;
                let timeoutId = null;

                const finish = option => {
                    if (completed) return;
                    completed = true;
                    observer?.disconnect();
                    if (intervalId) clearInterval(intervalId);
                    if (timeoutId) clearTimeout(timeoutId);
                    if (option) {
                        warehouseDebug && console.log('[debug][2026-07-23][货架下拉] option-captured', JSON.stringify({
                            shelf,
                            optionClass: option.className,
                            optionText: option.textContent.trim(),
                            optionPath: getElementCssPath(option, pageDocument),
                            optionHtml: option.outerHTML.slice(0, 500)
                        }));
                    }
                    resolve(option || null);
                };

                const check = () => {
                    const option = findVisibleShelfOption(shelf, pageWindow);
                    if (option) finish(option);
                };

                observer = new pageWindow.MutationObserver(check);
                observer.observe(pageDocument.body, {
                    childList: true,
                    subtree: true,
                    attributes: true,
                    attributeFilter: ['class', 'style'],
                    characterData: true
                });
                intervalId = setInterval(check, 25);
                timeoutId = setTimeout(() => finish(null), timeout);
                check();
            });
        }

        function getShelfSelectionState(row, pageWindow = getPageWindow()) {
            const select = row.querySelector('td.shelf_info td.new_shelf_el.list_container .ant-select');

            const placeholder = select?.querySelector('.ant-select-selection__placeholder');
            const searchInput = select?.querySelector('input.ant-select-search__field');

                open: !!select?.classList.contains('ant-select-open'),
                selectedText: selectedValue?.textContent.trim() || '',
                placeholderText: placeholder?.textContent.trim() || '',
                searchValue: searchInput?.value || '',
                cellText: row.querySelector('td.shelf_info td.new_shelf_el.list_container')?.textContent.trim() || '',
                activePath: getElementCssPath(pageWindow.document.activeElement, pageWindow.document)
            };
        }

        function getShelfOptionClickTarget(option) {

        function dispatchShelfSelectionEvents(option, pageWindow = getPageWindow()) {
            if (!option) return [];
            const child = option.classList?.contains('option')
                ? option
                : option.querySelector?.('.option');
            const targets = [child, option, option.parentElement]
                .filter(Boolean)
                .filter((target, index, list) => list.indexOf(target) === index);
            const dispatched = [];
            targets.forEach(target => {
                const rect = target.getBoundingClientRect();
                const eventInit = {
                    bubbles: true,
                    cancelable: true,
                    view: pageWindow,
                    detail: 1,
                    button: 0,
                    buttons: 1,
                    clientX: rect.left + Math.max(1, rect.width / 2),
                    clientY: rect.top + Math.max(1, rect.height / 2)
                };
                try {
                    const PointerCtor = pageWindow.PointerEvent || pageWindow.MouseEvent;
                    target.dispatchEvent(new PointerCtor('pointerover', eventInit));
                    target.dispatchEvent(new pageWindow.MouseEvent('mouseover', eventInit));
                    target.dispatchEvent(new PointerCtor('pointermove', eventInit));
                    target.dispatchEvent(new pageWindow.MouseEvent('mousemove', eventInit));
                    target.dispatchEvent(new PointerCtor('pointerdown', eventInit));
                    target.dispatchEvent(new pageWindow.MouseEvent('mousedown', eventInit));
                    target.dispatchEvent(new PointerCtor('pointerup', { ...eventInit, buttons: 0 }));
                    target.dispatchEvent(new pageWindow.MouseEvent('mouseup', { ...eventInit, buttons: 0 }));
                    target.dispatchEvent(new pageWindow.MouseEvent('click', { ...eventInit, buttons: 0 }));
                    dispatched.push({
                        path: getElementCssPath(target, pageWindow.document),
                        className: String(target.className || ''),
                        text: target.textContent.trim().slice(0, 120)
                    });
                } catch (error) {
                    warehouseDebug && console.warn('[debug][2026-07-23][货架选择探针] 事件分发失败', error);
                }
            });
            return dispatched;
        }

        function dispatchShelfKeyboardSelection(row, pageWindow = getPageWindow()) {
            const input = row.querySelector(
                'td.shelf_info td.new_shelf_el.list_container .ant-select-open input.ant-select-search__field'
            );
            if (!input) return false;
            try {
                input.focus();
                ['ArrowDown', 'Enter'].forEach(key => {
                    const init = {
                        bubbles: true,
                        cancelable: true,
                        key,
                        code: key,
                        keyCode: key === 'Enter' ? 13 : 40,
                        which: key === 'Enter' ? 13 : 40
                    };
                    input.dispatchEvent(new pageWindow.KeyboardEvent('keydown', init));
                    input.dispatchEvent(new pageWindow.KeyboardEvent('keyup', init));
                });
                return true;
            } catch (error) {
                warehouseDebug && console.warn('[debug][2026-07-23][货架选择探针] 键盘事件失败', error);
                return false;
            }
        }

        async function clickShelfOptionAndVerify(option, row, shelf, pageWindow = getPageWindow()) {
            const normalizedShelf = normalizeShelfText(shelf);
            const sku = getModalRowSku(row);
            let lastProbeSignature = '';
            const isCommitted = () => {
                const state = getShelfSelectionState(row, pageWindow);
                const selectedValueMatches = normalizeShelfText(state.selectedText) === normalizedShelf;
                const closedCellMatches = !state.open
                    && !state.searchValue
                    && normalizeShelfText(state.cellText) === normalizedShelf;
                const probeSignature = JSON.stringify({
                    open: state.open,
                    selectedText: state.selectedText,
                    searchValue: state.searchValue,
                    cellText: state.cellText,
                    activePath: state.activePath
                });
                if (selectedValueMatches || closedCellMatches) {
                    warehouseDebug && console.log('[debug][2026-07-23][货架提交判定] committed', JSON.stringify({
                        sku,
                        shelf,
                        reason: selectedValueMatches ? 'selected-value' : 'closed-cell',
                        state
                    }));
                    return state;
                }
                if (probeSignature !== lastProbeSignature) {
                    lastProbeSignature = probeSignature;
                    warehouseDebug && console.log('[debug][2026-07-23][货架提交判定] pending', JSON.stringify({ sku, shelf, state }));
                }
                return null;
            };
            const before = getShelfSelectionState(row, pageWindow);
            const clickTarget = getShelfOptionClickTarget(option);
            const dispatched = dispatchShelfSelectionEvents(option, pageWindow);
            warehouseDebug && console.log('[debug][2026-07-23][货架选择探针] pointer-mouse', JSON.stringify({
                shelf,
                optionPath: getElementCssPath(option, pageWindow.document),
                clickTargetPath: getElementCssPath(clickTarget, pageWindow.document),
                optionClass: String(option?.className || ''),
                clickTargetClass: String(clickTarget?.className || ''),
                dispatched
            }));
            let committed = await waitForElement(isCommitted, 500, 25);
            let fallbackTarget = clickTarget;
            if (!committed) {
                fallbackTarget = clickTarget?.isConnected
                    ? clickTarget
                    : getShelfOptionClickTarget(findVisibleShelfOption(shelf, pageWindow));
                if (fallbackTarget) {
                    warehouseDebug && console.log('[debug][2026-07-23][货架下拉] option-click-fallback', JSON.stringify({
                        shelf,
                        before,
                        optionPath: getElementCssPath(clickTarget, pageWindow.document),
                        optionConnected: clickTarget?.isConnected,
                        fallbackClass: fallbackTarget.className,
                        fallbackText: fallbackTarget.textContent.trim(),
                        fallbackPath: getElementCssPath(fallbackTarget, pageWindow.document)
                    }));
                    const fallbackEvents = dispatchShelfSelectionEvents(fallbackTarget, pageWindow);
                    warehouseDebug && console.log('[debug][2026-07-23][货架选择探针] fallback-pointer-mouse', JSON.stringify({
                        shelf,
                        fallbackPath: getElementCssPath(fallbackTarget, pageWindow.document),
                        fallbackEvents
                    }));
                    committed = await waitForElement(isCommitted, 1000, 25);
                }
            }
            if (!committed) {
                const keyboardDispatched = dispatchShelfKeyboardSelection(row, pageWindow);
                committed = await waitForElement(isCommitted, 1000, 25);
                warehouseDebug && console.log('[debug][2026-07-23][货架选择探针] keyboard', JSON.stringify({
                    shelf,
                    keyboardDispatched,
                    committed: !!committed,
                    state: getShelfSelectionState(row, pageWindow)
                }));
            }
            const after = getShelfSelectionState(row, pageWindow);
            warehouseDebug && console.log('[debug][2026-07-23][货架下拉] option-click-result', JSON.stringify({
                sku,
                shelf,
                committed: !!committed,
                before,
                after,
                optionPath: getElementCssPath(clickTarget, pageWindow.document),
                optionConnected: clickTarget?.isConnected,
                fallbackConnected: !!fallbackTarget?.isConnected
            }));
            return !!committed;
        }

        async function selectShelfForRow(row, shelf) {
            const pageWindow = getPageWindow();
            const shelfCell = row.querySelector('td.shelf_info td.new_shelf_el.list_container');
            const trigger = shelfCell?.querySelector('.placeholder_txt > span')
                || shelfCell?.querySelector('.placeholder_txt')
                || shelfCell?.querySelector('.ant-select-selection')
                || shelfCell?.querySelector('.ant-select');
            warehouseDebug && console.log('[debug][2026-07-23][货架下拉] trigger', JSON.stringify({
                shelf,
                found: !!trigger,
                triggerClass: trigger?.className || null,
                shelfCellClass: shelfCell?.className || null,
                rowText: row.textContent.trim().slice(0, 240),
                triggerHtml: trigger?.outerHTML?.slice(0, 500) || null
            }));
            if (!trigger) return false;

            row.scrollIntoView({ block: 'center', inline: 'nearest' });
            await new Promise(resolve => setTimeout(resolve, 80));
            dispatchHoverEvents(trigger, pageWindow);
            clickPageElement(trigger);
            const searchInput = await waitForElement(() => row.querySelector(
                'td.shelf_info td.new_shelf_el.list_container .ant-select-open input.ant-select-search__field'
            ), 1500, 50);
            warehouseDebug && console.log('[debug][2026-07-23][货架下拉] search-input', JSON.stringify({
                shelf,
                found: !!searchInput,
                path: getElementCssPath(searchInput, pageWindow.document),
                className: searchInput?.className || null,
                value: searchInput?.value || ''
            }));
            const optionPromise = waitForShelfOption(shelf, pageWindow, 5000);
            if (searchInput) setInputValue(searchInput, shelf);

            let option = await optionPromise;
            warehouseDebug && console.log('[debug][2026-07-23][货架下拉] after-open', JSON.stringify({
                shelf,
                optionFound: !!option,
                dropdowns: Array.from(pageWindow.document.querySelectorAll(
                    '.ant-select-dropdown, .ant-dropdown, .bs-antd-select-dropdown'
                )).map(dropdown => ({
                    className: dropdown.className,
                    visible: isVisibleElement(dropdown, pageWindow),
                    text: dropdown.textContent.trim().slice(0, 500),
                    options: Array.from(dropdown.querySelectorAll('li, [role="option"], .ant-dropdown-menu-item'))
                        .map(item => ({ className: item.className, text: item.textContent.trim() }))
                        .slice(0, 30)
                }))
                , visibleShelfNodes: describeVisibleShelfNodes(shelf, pageWindow)
            }));
            if (!option) {
                const fallbackTrigger = shelfCell.querySelector('.ant-select-selection, .ant-select') || trigger;
                warehouseSearchDebug && console.log('[DEBUG-HB13][dropdown-retry]', {
                    shelf,
                    sku: getModalRowSku(row),
                    hadSearchInput: !!searchInput,
                    fallbackPath: getElementCssPath(fallbackTrigger, pageWindow.document)
                });
                dispatchMouseClick(fallbackTrigger, pageWindow);
                const fallbackSearchInput = await waitForElement(() => row.querySelector(
                    'td.shelf_info td.new_shelf_el.list_container .ant-select-open input.ant-select-search__field'
                ), 1500, 50);
                if (fallbackSearchInput) setInputValue(fallbackSearchInput, shelf);
                option = await waitForShelfOption(shelf, pageWindow, 5000);
            }
            if (!option) {
                warehouseDebug && console.log('[debug][2026-07-23][货架下拉] option-missing', JSON.stringify({
                    shelf,
                    normalizedShelf: normalizeShelfText(shelf),
                    visibleText: Array.from(pageWindow.document.querySelectorAll(
                        '.ant-select-dropdown, .ant-dropdown, .bs-antd-select-dropdown'
                    )).filter(dropdown => isVisibleElement(dropdown, pageWindow))
                        .map(dropdown => dropdown.textContent.trim().slice(0, 500)),
                    visibleShelfNodes: describeVisibleShelfNodes(shelf, pageWindow)
                }));
                return false;
            }

            const selected = await clickShelfOptionAndVerify(option, row, shelf, pageWindow);
            warehouseDebug && console.log('[debug][2026-07-23][货架下拉] selected', JSON.stringify({
                shelf,
                selected,
                cellText: shelfCell?.textContent.trim() || null,
                optionClass: option.className,
                optionText: option.textContent.trim()
            }));
            return selected;
        }

        async function fillModalShelfLocations(skuList, shelfList) {
            const pageWindow = getPageWindow();
            const rows = await waitForElement(() => {
                const currentRows = getModalRows(pageWindow);
                return currentRows.length ? currentRows : null;
            }, 10000);
            if (!rows) return { ok: false, message: '未找到设置货架位弹窗内容' };
            attachShelfDropdownDebugListener(pageWindow);

            warehouseDebug && console.log('[debug][2026-07-23][货架下拉] modal-rows', JSON.stringify({
                rowCount: rows.length,
                rowSkus: rows.map(getModalRowSku),
                requestedSkus: skuList,
                requestedShelves: shelfList
            }));

            const rowBySku = new Map(rows.map(row => [getModalRowSku(row), row]));
            const unchanged = [];
            const updatePairs = [];
            for (let i = 0; i < skuList.length; i += 1) {
                const row = rowBySku.get(skuList[i]);
                if (!row) return { ok: false, message: `弹窗中未找到 SKU：${skuList[i]}` };
                const currentShelf = normalizeShelfText(row.querySelector('td.shelf_el')?.textContent);
                if (currentShelf && currentShelf === normalizeShelfText(shelfList[i])) {
                    unchanged.push({ sku: skuList[i], shelf: shelfList[i], currentShelf });
                    continue;
                }
                updatePairs.push({ sku: skuList[i], shelf: shelfList[i], row });
            }

            warehouseSearchDebug && console.log('[DEBUG-HB11][modal-preflight]', {
                modalSkus: rows.map(getModalRowSku),
                unchanged: unchanged.map(pair => ({ sku: pair.sku, current: pair.currentShelf, target: pair.shelf })),
                updates: updatePairs.map(pair => ({
                    sku: pair.sku,
                    current: normalizeShelfText(pair.row.querySelector('td.shelf_el')?.textContent),
                    target: pair.shelf
                }))
            });

            if (unchanged.length) {
                return {
                    ok: true,
                    requiresReselect: true,
                    unchanged,
                    updateSkuList: updatePairs.map(pair => pair.sku),
                    updateShelfList: updatePairs.map(pair => pair.shelf),
                    updatedCount: 0
                };
            }

            const failed = [];
            const selectedPairs = [];
            for (const pair of updatePairs) {
                if (!await selectShelfForRow(pair.row, pair.shelf)) {
                    warehouseSearchDebug && console.log('[DEBUG-HB11][select-failed]', {
                        sku: pair.sku,
                        currentShelf: normalizeShelfText(pair.row.querySelector('td.shelf_el')?.textContent),
                        targetShelf: pair.shelf,
                        rowHtml: pair.row.outerHTML.slice(0, 3000),
                        visibleDropdowns: Array.from(pageWindow.document.querySelectorAll(
                            '.ant-select-dropdown, .ant-dropdown, .bs-antd-select-dropdown'
                        )).filter(dropdown => isVisibleElement(dropdown, pageWindow)).map(dropdown => ({
                            className: dropdown.className,
                            text: dropdown.textContent.trim().slice(0, 1000),
                            html: dropdown.outerHTML.slice(0, 3000)
                        }))
                    });
                    failed.push({
                        sku: pair.sku,
                        shelf: pair.shelf,
                        currentShelf: normalizeShelfText(pair.row.querySelector('td.shelf_el')?.textContent),
                        reason: 'shelf-option-not-selected'
                    });
                    closeOpenShelfDropdown(pair.row, pageWindow);
                    continue;
                }
                selectedPairs.push(pair);
            }
            warehouseSearchDebug && failed.length && console.log('[debug][2026-07-23][货架失败项已跳过]', {
                failed,
                selected: selectedPairs.map(pair => ({ sku: pair.sku, shelf: pair.shelf }))
            });
            return {
                ok: true,
                requiresReselect: unchanged.length > 0 || failed.length > 0,
                unchanged,
                failed,
                updateSkuList: selectedPairs.map(pair => pair.sku),
                updateShelfList: selectedPairs.map(pair => pair.shelf),
                updatedCount: selectedPairs.length
            };
        }

        async function processShelfBatch(batch, batchNumber, totalBatches, resultEl) {
            const { skuList, shelfList } = batch;
            let submissionStarted = false;
            const fail = message => ({ ok: false, message, retryable: !submissionStarted });
            resultEl.innerHTML = `<span style="color:#1890ff;">正在处理第 ${batchNumber}/${totalBatches} 批（${skuList.length} 个 SKU）...</span>`;
            const searchTextarea = await openBatchSearch(resultEl);
            if (!searchTextarea) return fail('批量输入框未展开');

            setTextareaValue(searchTextarea, skuList.join('\n'));
            const searchContainer = searchTextarea.closest('.bs_antd_input_search_new');
            const searchBtn = Array.from(searchContainer?.querySelectorAll('.search_btn, i.bsicon_search1') || [])
                .find(button => getPageWindow().getComputedStyle(button).display !== 'none')
                || searchContainer?.querySelector('i.bsicon_search1');
            if (!searchBtn) return fail('未找到搜索按钮');

            warehouseSearchDebug && console.log('[DEBUG-HB9][search] input-set', {
                skuCount: skuList.length,
                firstSku: skuList[0] || null,
                lastSku: skuList[skuList.length - 1] || null,
                textareaValue: searchTextarea.value,
                textareaPath: getElementCssPath(searchTextarea, getPageWindow().document)
            });
            await clickShelfSearchButton(searchBtn);
            resultEl.innerHTML = `<span style="color:#1890ff;">第 ${batchNumber}/${totalBatches} 批正在等待搜索结果...</span>`;
            let resultsReady = await waitForSearchResults(skuList);
            warehouseSearchDebug && console.log('[DEBUG-HB9][search] wait-result', {
                ready: !!resultsReady,
                rowCount: getPageWindow().document.querySelectorAll('table.list_items > tbody > tr.list_item').length,
                textareaValue: searchTextarea.value
            });
            if (!resultsReady) return fail(`第 ${batchNumber} 批搜索结果未完整加载`);

            const searchPairs = getShelfSearchPairs(skuList, shelfList);
            warehouseSearchDebug && console.log('[DEBUG-HB10][shelf-compare]', {
                needsUpdate: searchPairs.available.map(pair => ({ sku: pair.sku, current: pair.currentShelf, target: pair.shelf })),
                unchanged: searchPairs.unchanged.map(pair => ({ sku: pair.sku, shelf: pair.shelf })),
                missing: searchPairs.missing.map(pair => pair.sku)
            });
            warehouseSearchDebug && console.log('[DEBUG-HB12][search-shelf-compare]', JSON.stringify({
                needsUpdate: searchPairs.available.map(pair => ({ sku: pair.sku, currentShelf: pair.currentShelf, targetShelf: pair.shelf })),
                unchanged: searchPairs.unchanged.map(pair => ({ sku: pair.sku, currentShelf: pair.currentShelf, targetShelf: pair.shelf })),
                missing: searchPairs.missing.map(pair => ({ sku: pair.sku, targetShelf: pair.shelf }))
            }));
            if (!searchPairs.available.length) {
                return { ok: true, skipped: true, missing: searchPairs.missing, unchanged: searchPairs.unchanged };
            }

            if (!await ensureShelfPageSize()) {
                return fail(`第 ${batchNumber} 批未能切换为 ${SHELF_PAGE_SIZE} 条/页`);
            }
            resultsReady = await waitForSearchResults(skuList);
            if (!resultsReady) return fail(`第 ${batchNumber} 批切换分页后搜索结果未完整加载`);

            const availableSkuList = searchPairs.available.map(pair => pair.sku);
            const availableShelfList = searchPairs.available.map(pair => pair.shelf);
            if (!await selectResultRowsForSkus(availableSkuList)) {
                return fail(`第 ${batchNumber} 批未能仅勾选需要修改的 SKU`);
            }
            if (!await clickBatchShelfAction()) {
                return fail(`第 ${batchNumber} 批未找到批量绑定货架号操作`);
            }

            let fillResult = await fillModalShelfLocations(availableSkuList, availableShelfList);
            if (!fillResult.ok) return fail(fillResult.message);

            let unchangedPairs = searchPairs.unchanged.concat(fillResult.unchanged || []);
            let skippedShelfPairs = [...(fillResult.failed || [])];
            // 选择失败的 SKU 不再阻断整批；关闭预检弹窗后，只重新勾选尚未失败的 SKU。
            for (let reselectAttempt = 0; fillResult.requiresReselect; reselectAttempt += 1) {
                if (reselectAttempt >= 3) {
                    return fail(`第 ${batchNumber} 批货架位选择失败项过多，已跳过 ${skippedShelfPairs.length} 个但仍无法稳定重试`);
                }
                if (!await closeShelfModalWithoutSubmitting()) {
                    return fail(`第 ${batchNumber} 批未能关闭预检货架位弹窗`);
                }
                if (!fillResult.updateSkuList.length) {
                    return {
                        ok: true,
                        skipped: true,
                        missing: searchPairs.missing,
                        unchanged: unchangedPairs,
                        unverifiedItems: skippedShelfPairs.map(item => ({
                            sku: item.sku,
                            actualShelf: item.currentShelf,
                            targetShelf: item.shelf
                        }))
                    };
                }
                warehouseSearchDebug && console.log('[debug][2026-07-23][货架失败项重选剩余SKU]', {
                    batchNumber,
                    reselectAttempt: reselectAttempt + 1,
                    skippedShelfPairs,
                    remainingSkuList: fillResult.updateSkuList,
                    remainingShelfList: fillResult.updateShelfList
                });
                if (!await selectResultRowsForSkus(fillResult.updateSkuList)) {
                    return fail(`第 ${batchNumber} 批未能仅勾选未失败的 SKU`);
                }
                if (!await clickBatchShelfAction()) {
                    return fail(`第 ${batchNumber} 批未能重新打开设置货架位弹窗`);
                }
                fillResult = await fillModalShelfLocations(fillResult.updateSkuList, fillResult.updateShelfList);
                if (!fillResult.ok) return fail(fillResult.message);
                unchangedPairs = unchangedPairs.concat(fillResult.unchanged || []);
                skippedShelfPairs = skippedShelfPairs.concat(fillResult.failed || []);
            }
            if (!fillResult.updatedCount) {
                if (!await closeShelfModalWithoutSubmitting()) {
                    return fail(`第 ${batchNumber} 批无需修改，但未能关闭货架位弹窗`);
                }
                return {
                    ok: true,
                    skipped: true,
                    missing: searchPairs.missing,
                    unchanged: unchangedPairs,
                    unverifiedItems: skippedShelfPairs.map(item => ({
                        sku: item.sku,
                        actualShelf: item.currentShelf,
                        targetShelf: item.shelf
                    }))
                };
            }

            warehouseSearchDebug && console.log('[debug][2026-07-23][货架提交前数量]', {
                batchNumber,
                searchedCount: skuList.length,
                missingCount: searchPairs.missing.length,
                unchangedCount: unchangedPairs.length,
                expectedProcessedCount: fillResult.updatedCount,
                submitSkuList: fillResult.updateSkuList,
                submitShelfList: fillResult.updateShelfList
            });
            const unchangedMessage = unchangedPairs.length ? `，${unchangedPairs.length} 个无需修改` : '';
            resultEl.innerHTML = `<span style="color:#1890ff;">第 ${batchNumber}/${totalBatches} 批已填入，本次提交 ${fillResult.updatedCount} 个，正在自动提交${unchangedMessage}...</span>`;
            submissionStarted = true;
            const confirmation = await autoConfirmShelfModal(fillResult.updatedCount);
            if (!confirmation.confirmed) {
                const reportedText = confirmation.reportedProcessedCount == null
                    ? '未读到成功/失败数'
                    : `成功+失败=${confirmation.reportedProcessedCount}`;
                return fail(`第 ${batchNumber} 批结果未核对通过：应处理 ${fillResult.updatedCount} 个，${reportedText}（${confirmation.action}），结果弹窗和断点已保留`);
            }
            warehouseSearchDebug && console.log('[debug][2026-07-23][货架批次已核对]', {
                batchNumber,
                expectedProcessedCount: fillResult.updatedCount,
                reportedSuccess: confirmation.success,
                reportedFailure: confirmation.failure,
                reportedProcessedCount: confirmation.reportedProcessedCount,
                resultCountMatches: confirmation.resultCountMatches
            });
            // 结果详情是 BigSeller 的接口报告，不作为最终结果；提交后主列表的实际货架位才是兜底依据。
            const submittedSkuList = fillResult.updateSkuList;
            const submittedShelfList = fillResult.updateShelfList;
            const verification = await verifyInventoryShelfPositions(submittedSkuList, submittedShelfList);
            const unverifiedItems = verification.details.filter(item => (
                normalizeShelfText(item.actualShelf) !== normalizeShelfText(item.targetShelf)
            ));
            skippedShelfPairs.forEach(item => {
                if (!unverifiedItems.some(existing => existing.sku === item.sku)) {
                    unverifiedItems.push({
                        sku: item.sku,
                        actualShelf: item.currentShelf,
                        targetShelf: item.shelf
                    });
                }
            });
            const reportedFailureSkuSet = new Set((confirmation.reportedFailureSkuCandidates || [])
                .map(normalizeShelfText));
            const resultDetailComparison = verification.details.map(item => ({
                ...item,
                reportedFailed: reportedFailureSkuSet.has(normalizeShelfText(item.sku)),
                actuallyApplied: normalizeShelfText(item.actualShelf) === normalizeShelfText(item.targetShelf)
            }));
            warehouseSearchDebug && console.log('[debug][2026-07-23][货架详情与实际货架位兜底]', {
                batchNumber,
                reportedFailureSkuCandidates: confirmation.reportedFailureSkuCandidates || [],
                detailLines: confirmation.detailLines || [],
                reportedFailedButActuallyApplied: resultDetailComparison.filter(item => (
                    item.reportedFailed && item.actuallyApplied
                )),
                actualFailedButNotReported: resultDetailComparison.filter(item => (
                    !item.reportedFailed && !item.actuallyApplied
                )),
                actualUnverifiedItems: unverifiedItems
            });
            return {
                ok: true,
                skipped: false,
                missing: searchPairs.missing,
                unchanged: unchangedPairs,
                unverifiedItems
            };
        }

        async function closeShelfModalForRetry() {
            const pageWindow = getPageWindow();
            const pageDocument = pageWindow.document;
            const modal = Array.from(pageDocument.querySelectorAll('.ant-modal-wrap'))
                .filter(candidate => isVisibleElement(candidate, pageWindow))
                .find(candidate => candidate.querySelector('.page_list > table > tbody'));
            if (!modal) return true;
            const closeButton = modal.querySelector('.ant-modal-close, .ant-modal-header .close, .ant-modal-content > button');
            if (!closeButton) return false;
            clickPageElement(closeButton);
            let closed = await waitForElement(() => (
                !modal.isConnected || !isVisibleElement(modal, pageWindow) ? pageDocument.body : null
            ), 2000, 50);
            if (!closed) {
                dispatchMouseClick(closeButton, pageWindow);
                closed = await waitForElement(() => (
                    !modal.isConnected || !isVisibleElement(modal, pageWindow) ? pageDocument.body : null
                ), 2000, 50);
            }
            return !!closed;
        }

        async function processShelfBatchWithRetry(batch, batchNumber, totalBatches, resultEl) {
            const maxAttempts = 3;
            for (let attempt = 1; attempt <= maxAttempts; attempt += 1) {
                let result;
                try {
                    result = await processShelfBatch(batch, batchNumber, totalBatches, resultEl);
                } catch (error) {
                    console.error('[货架号绑定] 提交前批次异常:', error);
                    return { ok: false, message: `第 ${batchNumber} 批发生异常，未自动重试以避免重复提交` };
                }
                if (result.ok || !result.retryable || attempt === maxAttempts) return result;

                const modalClosed = await closeShelfModalForRetry();
                warehouseSearchDebug && console.log('[debug][2026-07-23][货架批次自动重试]', {
                    batchNumber,
                    attempt,
                    nextAttempt: attempt + 1,
                    failureMessage: result.message,
                    modalClosed
                });
                resultEl.innerHTML = `<span style="color:#fa8c16;">第 ${batchNumber}/${totalBatches} 批遇到问题，已关闭弹窗，正在第 ${attempt + 1}/3 次重试...</span>`;
                await new Promise(resolve => setTimeout(resolve, 500));
            }
            return { ok: false, message: `第 ${batchNumber} 批重试次数已用尽` };
        }

        async function verifyAllShelfBindings(skuList, shelfList, resultEl) {
            if (!await ensureShelfPageSize()) {
                return { ok: false, message: `最终复核未能切换为 ${SHELF_PAGE_SIZE} 条/页` };
            }
            const batches = splitShelfBatches(skuList, shelfList);
            const missingItems = [];
            const unverifiedItems = [];
            for (let index = 0; index < batches.length; index += 1) {
                const batch = batches[index];
                const batchNumber = index + 1;
                resultEl.innerHTML = `<span style="color:#1890ff;">全部提交完成，正在最终复核第 ${batchNumber}/${batches.length} 批（${batch.skuList.length} 个 SKU）...</span>`;
                const searchTextarea = await openBatchSearch(resultEl);
                if (!searchTextarea) return { ok: false, message: '最终复核未能展开批量输入框' };

                setTextareaValue(searchTextarea, batch.skuList.join('\n'));
                const searchContainer = searchTextarea.closest('.bs_antd_input_search_new');
                const searchButton = Array.from(searchContainer?.querySelectorAll('.search_btn, i.bsicon_search1') || [])
                    .find(button => getPageWindow().getComputedStyle(button).display !== 'none')
                    || searchContainer?.querySelector('i.bsicon_search1');
                if (!searchButton) return { ok: false, message: '最终复核未找到搜索按钮' };
                await clickShelfSearchButton(searchButton);
                if (!await waitForSearchResults(batch.skuList)) {
                    return { ok: false, message: `最终复核第 ${batchNumber} 批搜索结果未加载` };
                }

                const pairs = getShelfSearchPairs(batch.skuList, batch.shelfList);
                missingItems.push(...pairs.missing);
                unverifiedItems.push(...pairs.available.map(pair => ({
                    sku: pair.sku,
                    actualShelf: pair.currentShelf,
                    targetShelf: pair.shelf
                })));
                warehouseSearchDebug && console.log('[debug][2026-07-23][货架最终复核]', {
                    batchNumber,
                    totalBatches: batches.length,
                    checkedCount: batch.skuList.length,
                    matchedCount: pairs.unchanged.length,
                    missingItems: pairs.missing,
                    unverifiedItems: pairs.available.map(pair => ({
                        sku: pair.sku,
                        actualShelf: pair.currentShelf,
                        targetShelf: pair.shelf
                    }))
                });
            }
            return { ok: true, missingItems, unverifiedItems };
        }

        // 绑定按钮：批量处理 SKU 和货架号
        document.getElementById('c_shelf_confirm').addEventListener('click', async () => {
            if (workflowRunning) return;
            workflowRunning = true;
            const confirmBtn = document.getElementById('c_shelf_confirm');
            confirmBtn.disabled = true;
            try {
                const skuText = document.getElementById('c_shelf_sku').value.trim();
                const shelfText = document.getElementById('c_shelf_location').value.trim();
                const resultEl = document.getElementById('c_shelf_result');
                if (!skuText || !shelfText) {
                    resultEl.innerHTML = '<span style="color:#ff4d4f;">请填写 SKU 和货架号</span>';
                    return;
                }

                const skuList = skuText.split(/[\n\r]+/).map(s => s.trim()).filter(s => s);
                const shelfList = shelfText.split(/[\n\r]+/).map(s => s.trim()).filter(s => s);
                if (skuList.length !== shelfList.length) {
                    resultEl.innerHTML = '<span style="color:#ff4d4f;">SKU 和货架号数量不一致</span>';
                    return;
                }

                let job = getOrCreateShelfJob(skuList, shelfList);
                job.missingItems ||= [];
                job.failedItems ||= [];
                if (job.completedCount >= skuList.length && job.finalVerification?.completed) {
                    clearShelfJob();
                    job = getOrCreateShelfJob(skuList, shelfList);
                }
                writeShelfJob(job);

                const batches = splitShelfBatches(skuList, shelfList, job.completedCount);

                for (let index = 0; index < batches.length; index += 1) {
                    const batch = batches[index];
                    const batchNumber = Math.floor(batch.start / SHELF_BATCH_SIZE) + 1;
                    const totalBatches = Math.ceil(skuList.length / SHELF_BATCH_SIZE);
                    const batchResult = await processShelfBatchWithRetry(batch, batchNumber, totalBatches, resultEl);
                    if (!batchResult.ok) {
                        resultEl.innerHTML = `<span style="color:#ff4d4f;">${batchResult.message}</span>`;
                        return;
                    }
                    const missingKeys = new Set(job.missingItems.map(item => `${item.sku}\u0000${item.shelf}`));
                    (batchResult.missing || []).forEach(item => {
                        const key = `${item.sku}\u0000${item.shelf}`;
                        if (!missingKeys.has(key)) {
                            missingKeys.add(key);
                            job.missingItems.push(item);
                        }
                    });
                    const failedKeys = new Set(job.failedItems.map(item => `${item.sku}\u0000${item.targetShelf}`));
                    (batchResult.unverifiedItems || []).forEach(item => {
                        const key = `${item.sku}\u0000${item.targetShelf}`;
                        if (!failedKeys.has(key)) {
                            failedKeys.add(key);
                            job.failedItems.push(item);
                        }
                    });
                    job.completedCount = batch.end;
                    writeShelfJob(job);
                    const batchMissingCount = (batchResult.missing || []).length;
                    const batchUnchangedCount = (batchResult.unchanged || []).length;
                    const batchUnverifiedCount = (batchResult.unverifiedItems || []).length;
                    const missingMessage = batchMissingCount ? `；本批未找到 ${batchMissingCount} 个` : '';
                    const unchangedMessage = batchUnchangedCount ? `；${batchUnchangedCount} 个已是目标货架位` : '';
                    const unverifiedMessage = batchUnverifiedCount ? `；待核对 ${batchUnverifiedCount} 个` : '';
                    const batchStatus = batchResult.skipped ? '已跳过' : '已确认';
                    resultEl.innerHTML = `<span style="color:#52c41a;">第 ${batchNumber}/${totalBatches} 批${batchStatus}，已完成 ${job.completedCount}/${skuList.length} 个 SKU${unchangedMessage}${missingMessage}${unverifiedMessage}</span>`;
                }
                job.finalVerification = { completed: false, startedAt: new Date().toISOString() };
                writeShelfJob(job);
                const finalVerification = await verifyAllShelfBindings(skuList, shelfList, resultEl);
                if (!finalVerification.ok) {
                    resultEl.innerHTML = `<span style="color:#ff4d4f;">${finalVerification.message}，最终复核断点已保留</span>`;
                    return;
                }
                // 最终结果仅以重新搜索到的实际货架位为准，覆盖过程中的临时接口结果。
                job.missingItems = finalVerification.missingItems;
                job.failedItems = finalVerification.unverifiedItems;
                job.finalVerification = {
                    completed: true,
                    completedAt: new Date().toISOString(),
                    missingCount: job.missingItems.length,
                    unverifiedCount: job.failedItems.length
                };
                writeShelfJob(job);
                const successCount = skuList.length - job.missingItems.length - job.failedItems.length;
                const missingSummary = job.missingItems.length ? `，成功 ${successCount} 个，未找到 ${job.missingItems.length} 个` : '';
                const failedSummary = job.failedItems.length ? `，待核对 ${job.failedItems.length} 个` : '';
                resultEl.innerHTML = `<span style="color:#52c41a;">全部 ${skuList.length} 个 SKU 已提交并完成最终复核${missingSummary}${failedSummary}，断点记录已保存</span>`;
            } catch (error) {
                console.error('[货架号绑定] 批量流程失败:', error);
                document.getElementById('c_shelf_result').innerHTML = '<span style="color:#ff4d4f;">批量流程执行失败，请查看控制台</span>';
            } finally {
                workflowRunning = false;
                confirmBtn.disabled = false;
            }
        });

        const savedShelfJob = readShelfJob();
        if (savedShelfJob?.skuList?.length && savedShelfJob?.shelfList?.length) {
            document.getElementById('c_shelf_sku').value = savedShelfJob.skuList.join('\n');
            document.getElementById('c_shelf_location').value = savedShelfJob.shelfList.join('\n');
            const total = savedShelfJob.skuList.length;
            const completed = Math.min(savedShelfJob.completedCount || 0, total);
            const missingCount = Array.isArray(savedShelfJob.missingItems) ? savedShelfJob.missingItems.length : 0;
            const failedCount = Array.isArray(savedShelfJob.failedItems) ? savedShelfJob.failedItems.length : 0;
            const finalVerified = savedShelfJob.finalVerification?.completed === true;
            document.getElementById('c_shelf_result').innerHTML = completed >= total && finalVerified
                ? `<span style="color:#52c41a;">上次任务已处理并最终复核 ${total} 个 SKU，成功 ${total - missingCount - failedCount} 个，未找到 ${missingCount} 个，待核对 ${failedCount} 个，可清除断点后重新开始</span>`
                : completed >= total
                    ? `<span style="color:#1890ff;">全部 ${total} 个 SKU 已提交，最终复核尚未完成，点击按钮继续复核</span>`
                : `<span style="color:#1890ff;">发现断点：已完成 ${completed}/${total} 个 SKU，未找到 ${missingCount} 个，待核对 ${failedCount} 个，点击按钮可从第 ${Math.floor(completed / SHELF_BATCH_SIZE) + 1} 批继续</span>`;
        }
    }

    // 根据当前页面初始化对应功能
    if (location.pathname.includes('/web/inventory/warehouseInventory.htm')) {
        // 等待页面渲染后初始化
        const whObserver = new MutationObserver(() => {
            if (document.body && !document.getElementById('c_shelf_panel')) {
                initWarehousePanel();
            }
        });
        whObserver.observe(document.documentElement, { childList: true, subtree: true });
        setTimeout(initWarehousePanel, 1000);
    }

})();
